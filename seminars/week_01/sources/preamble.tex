  \mathord{\tikz[baseline=(x.base)]{

% A red cross on an arrow. It is drawn in the coordinates of the arrow, so
% that both lines cross the arrow at about 65 degrees.
\newcommand{\edgecross}{%
  \draw[red, line width=0.7pt, -] (-1.7pt, -3.6pt) -- (1.7pt, 3.6pt);
  \draw[red, line width=0.7pt, -] (1.7pt, -3.6pt) -- (-1.7pt, 3.6pt);}

\newcommand{\crossgraph}[5]{%
  \begin{tikzpicture}[
      node distance=0.6cm and 0.55cm, >={Stealth},
      var/.style={circle, draw, minimum size=0.65cm, inner sep=1pt},
      f/.style={->, very thick},
    ]
    \node[var, draw=fP, very thick, src#1] (P) {$P$};
    \node[var, draw=fD, very thick, src#2, right=1.6cm of P] (D) {$D$};
    \node[var, nd#3, below left=of P] (H) {$H$};
    \node[var, nd#4, below right=of P] (A) {$A$};
    \node[var, nd#5, below=of A] (T) {$T$};
    \draw[f, fH, ed#3] (P) -- (H);
    \draw[f, fA, ed#4] (P) -- (A);
    \draw[f, fA, ed#4] (D) -- (A);
    \draw[f, fT, ed#5] (A) -- (T);
    \node[draw, fill=blue!15, minimum size=0.3cm, right=1.2cm of D] (key) {};
    \node[right=0.1cm of key] {variables of $p(G)$};
  \end{tikzpicture}%
}
