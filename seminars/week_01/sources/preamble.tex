\newcommand{\strike}[1]{%
  \mathord{\tikz[baseline=(x.base)]{
    \node[inner sep=0pt, outer sep=0pt] (x) {$#1$};
    \draw[red, thick] (x.south west) -- (x.north east);}}}

\newcommand{\crossgraph}[5]{%
  \begin{tikzpicture}[
      node distance=0.6cm and 0.55cm, >={Stealth},
      var/.style={circle, draw, minimum size=0.65cm, inner sep=1pt},
    ]
    \node[var, src#1] (P) {$P$};
    \node[var, src#2, right=1.6cm of P] (D) {$D$};
    \node[var, nd#3, below left=of P] (H) {$H$};
    \node[var, nd#4, below right=of P] (A) {$A$};
    \node[var, nd#5, below=of A] (T) {$T$};
    \draw[->, ed#3] (P) -- (H);
    \draw[->, ed#4] (P) -- (A);
    \draw[->, ed#4] (D) -- (A);
    \draw[->, ed#5] (A) -- (T);
    \node[draw, fill=blue!15, minimum size=0.3cm, right=1.2cm of D] (key) {};
    \node[right=0.1cm of key] {variables of $p(G)$};
  \end{tikzpicture}%
}
