Let $x_1 \sim \Poiss(\lambda_1)$ and $x_2 \sim \Poiss(\lambda_2)$ be two
independent Poisson random variables, that is,
$p(x_i = k) = \exp(-\lambda_i) \frac{\lambda_i^k}{k!}$,
$k = 0, 1, 2, \ldots$ Prove that
$x_1 + x_2 \sim \Poiss(\lambda_1 + \lambda_2)$.

\begin{solution}
\solpart{Idea}
For $y = f(x_1, \ldots, x_n)$ with discrete $x_i$, split the event
$\{y = k\}$ into disjoint events and sum:
\[
  p(y = k)
  = \sum_{f(j_1, \ldots, j_n) = k} p(x_1 = j_1, \ldots, x_n = j_n)
  = \sum_{f(j_1, \ldots, j_n) = k} \prod_{i} p(x_i = j_i)
  \quad \text{(if independent)}.
\]
Then simplify the sum and compare it with known distributions.

\solpart{Solution}
Goal: show that $p(x_1 + x_2 = k) = e^{-(\lambda_1 + \lambda_2)}
(\lambda_1 + \lambda_2)^k / k!$ for all $k \geq 0$.
\begin{align*}
  p(x_1 + x_2 = k)
  &= p\Bigl(\bigcup_{j=0}^{k} \{x_1 = j, \ x_2 = k - j\}\Bigr)
    && \text{($x_1, x_2 \geq 0$)} \\
  &= \sum_{j=0}^{k} p(x_1 = j, \ x_2 = k - j)
    && \text{(disjoint events)} \\
  &= \sum_{j=0}^{k} p(x_1 = j) \, p(x_2 = k - j)
    && \text{(independence)} \\
  &= \sum_{j=0}^{k} e^{-\lambda_1} \frac{\lambda_1^j}{j!} \cdot
     e^{-\lambda_2} \frac{\lambda_2^{k-j}}{(k-j)!}
    && \text{(Poisson probabilities)} \\
  &= e^{-\lambda_1} e^{-\lambda_2}
     \sum_{j=0}^{k} \frac{\lambda_1^j \lambda_2^{k-j}}{j! \, (k-j)!}
    && \text{(factors without $j$ out of the sum)} \\
  &= e^{-(\lambda_1 + \lambda_2)}
     \sum_{j=0}^{k} \frac{\lambda_1^j \lambda_2^{k-j}}{j! \, (k-j)!}
    && \text{($e^{x} e^{y} = e^{x + y}$)} \\
  &= \frac{e^{-(\lambda_1 + \lambda_2)}}{k!}
     \sum_{j=0}^{k} \frac{k!}{j! \, (k-j)!} \lambda_1^j \lambda_2^{k-j}
    && \text{(multiply and divide by $k!$)} \\
  &= \frac{e^{-(\lambda_1 + \lambda_2)}}{k!}
     \sum_{j=0}^{k} \binom{k}{j} \lambda_1^j \lambda_2^{k-j}
    && \textstyle\text{($\binom{k}{j} = \frac{k!}{j! \, (k-j)!}$)} \\
  &= \frac{e^{-(\lambda_1 + \lambda_2)}}{k!} (\lambda_1 + \lambda_2)^k
    && \text{(binomial theorem)}.
\end{align*}
This is $\Poiss(\lambda_1 + \lambda_2)$.

\solpart{Conclusion}
A sum of independent Poisson variables is a Poisson variable, and the
parameters add. By induction, this is true for any finite number of
variables.
\end{solution}
