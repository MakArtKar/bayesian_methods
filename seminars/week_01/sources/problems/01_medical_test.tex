\label{prob:medical-test}
A medical test finds a serious disease in a patient. The test is 99\%
accurate: if the patient has the disease, the test is positive with
probability 99\%; if the patient does not have the disease, the test is
negative with probability 99\%. The disease is rare: only one person in
10\,000 has it. Find the probability that the patient really has the
disease.

\begin{solution}
\solpart{Interpretation}
$D = 1$: the patient has the disease. $T = 1$: the test result is positive.
\[
  p(D = 1) = 10^{-4}, \qquad
  p(T = 1 \mid D = 1) = 0.99, \qquad
  p(T = 0 \mid D = 0) = 0.99.
\]
Find $p(D = 1 \mid T = 1)$.

\solpart{Idea}
The disease causes the test result. The causal graph has one arrow:
\begin{center}
  \begin{tikzpicture}[
      node distance=1.5cm, >={Stealth},
      var/.style={circle, draw, minimum size=0.75cm},
    ]
    \node[var] (D) {$D$};
    \node[var, right=of D] (T) {$T$};
    \draw[->] (D) -- node[above, font=\footnotesize] {$p(T \mid D)$} (T);
    \node[above=0.1cm of D, font=\footnotesize] {$p(D)$};
  \end{tikzpicture}
\end{center}
The problem gives the probability of the source node, $p(D)$, and the
probability along the arrow, $p(T \mid D)$.

\emph{Rules.} Each rule gives a formula for a probability that is not
given:
\begin{align*}
  &\text{R1. $p(D \mid T)$, against the arrow:}
    && p(D \mid T) = \frac{p(D, T)}{p(T)}, \\
  &\text{R2. $p(D, T)$, along the arrow:}
    && p(D, T) = p(D) \, p(T \mid D), \\
  &\text{R3. $p(T)$, the parent $D$ is not fixed:}
    && p(T) = \sum_{d} p(D = d, T), \\
  &\text{R4. The other value of a binary variable:}
    && p(D = 0) = 1 - p(D = 1), \\
  &&& p(T = 1 \mid D = 0) = 1 - p(T = 0 \mid D = 0).
\end{align*}
R1, R2, and R3 together give Bayes' theorem:
\[
  p(D \mid T)
  = \frac{p(D) \, p(T \mid D)}{\sum_{d} p(D = d) \, p(T \mid D = d)}.
\]
The same rules are true for any arrow $A \to B$: replace $D$ with $A$ and
$T$ with $B$. For a continuous parent, the sum in R3 is an integral.

\emph{Computation graph.} Put the goal at the top. For each node that is
not given, apply the rule for its form. Put the terms of the formula below
the node as new nodes. Stop when all nodes at the bottom are given. Then
calculate the nodes from the bottom to the top.

\solpart{Solution}
\emph{Computation graph.} Each arrow shows the rule and the formula for the
node above it. Shaded nodes are given.
\begin{center}
  \begin{tikzpicture}[
      x=1cm, y=1cm, font=\small, >={Stealth},
      val/.style={draw, rounded corners, inner sep=3pt},
      given/.style={val, fill=black!8},
      goal/.style={val, very thick},
      op/.style={font=\footnotesize, inner sep=2pt},
    ]
    % Goal.
    \node[goal] (G) at (1.5, 0.2) {$p(D{=}1 \mid T{=}1)$};
    % Marginal probability.
    \node[val] (T1) at (4.5, -2) {$p(T{=}1)$};
    % Joint probabilities.
    \node[val] (J1) at (1.5, -4) {$p(D{=}1, T{=}1)$};
    \node[val] (J0) at (7.5, -4) {$p(D{=}0, T{=}1)$};
    % Complements.
    \node[val] (D0)   at (6, -6) {$p(D{=}0)$};
    \node[val] (T1D0) at (9, -6) {$p(T{=}1 \mid D{=}0)$};
    % Given values.
    \node[given] (T1D1) at (0, -7.6) {$p(T{=}1 \mid D{=}1)$};
    \node[given] (D1)   at (4, -7.6) {$p(D{=}1)$};
    \node[given] (T0D0) at (9, -7.6) {$p(T{=}0 \mid D{=}0)$};
    % Arrows go from the inputs up to the node that uses them.
    \coordinate (g) at (1.5, -1);
    \draw (J1) -- (g);
    \draw (T1) -- (g);
    \draw[->] (g) -- node[op, left] {R1: $p(D{=}1, T{=}1) \,/\, p(T{=}1)$} (G);
    \coordinate (t1) at (4.5, -3);
    \draw (J1) -- (t1);
    \draw (J0) -- (t1);
    \draw[->] (t1) -- node[op, right] {R3: $p(D{=}1, T{=}1) + p(D{=}0, T{=}1)$} (T1);
    \coordinate (j1) at (1.5, -5);
    \draw (T1D1) -- (j1);
    \draw (D1) -- (j1);
    \draw[->] (j1) -- node[op, left] {R2: $p(D{=}1) \, p(T{=}1 \mid D{=}1)$} (J1);
    \coordinate (j0) at (7.5, -5);
    \draw (D0) -- (j0);
    \draw (T1D0) -- (j0);
    \draw[->] (j0) -- node[op, right] {R2: $p(D{=}0) \, p(T{=}1 \mid D{=}0)$} (J0);
    \draw[->] (D1) -- node[op, below right] {R4: $1 - p(D{=}1)$} (D0);
    \draw[->] (T0D0) -- node[op, right] {R4: $1 - p(T{=}0 \mid D{=}0)$} (T1D0);
  \end{tikzpicture}
\end{center}

\solstep{Step 1. Find $p(D = 0)$ with R4.}
\[
  p(D = 0) = 1 - p(D = 1) = 1 - 0.0001 = 0.9999.
\]

\solstep{Step 2. Find $p(T = 1 \mid D = 0)$ with R4.}
\[
  p(T = 1 \mid D = 0) = 1 - p(T = 0 \mid D = 0) = 1 - 0.99 = 0.01.
\]

\solstep{Step 3. Find $p(D = 1, T = 1)$ with R2.}
\[
  p(D = 1, T = 1)
  = p(D = 1) \, p(T = 1 \mid D = 1)
  = 0.0001 \cdot 0.99
  = 0.000099.
\]

\solstep{Step 4. Find $p(D = 0, T = 1)$ with R2.}
\[
  p(D = 0, T = 1)
  = p(D = 0) \, p(T = 1 \mid D = 0)
  = 0.9999 \cdot 0.01
  = 0.009999.
\]

\solstep{Step 5. Find $p(T = 1)$ with R3.}
\begin{align*}
  p(T = 1)
  &= \sum_{D} p(D, T = 1)
   = p(D = 1, T = 1) + p(D = 0, T = 1) \\
  &= 0.000099 + 0.009999
   = 0.010098.
\end{align*}

\solstep{Step 6. Find $p(D = 1 \mid T = 1)$ with R1.}
\begin{align*}
  p(D = 1 \mid T = 1)
  &= \frac{p(D = 1, T = 1)}{p(T = 1)}
   = \frac{0.000099}{0.010098}
   = \frac{99}{10098}
    && \text{(multiply by $10^6$)} \\
  &= \frac{99}{99 \cdot 102}
   = \frac{1}{102}
   \approx 0.0098.
\end{align*}

\solpart{Conclusion}
The probability increases from 0.01\% to only 1\%. In 10\,000 people, the
test finds $10\,000 \cdot 0.000099 \approx 1$ sick person and
$10\,000 \cdot 0.009999 \approx 100$ healthy people. Thus most positive
results of a mass test for a rare disease are false. A second positive
independent test uses $p(D = 1 \mid T_1 = 1) = 1/102$ as the prior:
\begin{align*}
  p(D = 1 \mid T_1 = 1, T_2 = 1)
  &= \frac{\frac{1}{102} \cdot 0.99}
          {\frac{1}{102} \cdot 0.99 + \bigl(1 - \frac{1}{102}\bigr) \cdot 0.01} \\
  &= \frac{\frac{1}{102} \cdot 0.99}
          {\frac{1}{102} \cdot 0.99 + \frac{101}{102} \cdot 0.01} \\
  &= \frac{0.99}{0.99 + 101 \cdot 0.01}
    && \text{(multiply by 102)} \\
  &= \frac{0.99}{0.99 + 1.01}
   = \frac{0.99}{2}
   = 0.495.
\end{align*}
\end{solution}
