\label{prob:medical-test}
A medical test finds a serious disease in a patient. The test is 99\%
accurate: if the patient has the disease, the test is positive with
probability 99\%; if the patient does not have the disease, the test is
negative with probability 99\%. The disease is rare: only one person in
10\,000 has it. Find the probability that the patient really has the
disease.

\begin{solution}
\solpart{Interpretation}
$D = 1$: the patient has the disease. $T = 1$: the test result is positive.
\begin{align*}
  p(D = 1) &= 10^{-4}, & p(D = 0) &= 1 - 10^{-4}, \\
  p(T = 1 \mid D = 1) &= 0.99, & p(T = 1 \mid D = 0) &= 1 - 0.99 = 0.01.
\end{align*}
Find $p(D = 1 \mid T = 1)$.

\solpart{Idea}
The disease causes the test result. The causal graph has one arrow:
\begin{center}
  \begin{tikzpicture}[
      node distance=1.5cm, >={Stealth},
      var/.style={circle, draw, minimum size=0.75cm},
    ]
    \node[var] (D) {$D$};
    \node[var, right=of D] (T) {$T$};
    \draw[->] (D) -- node[above, font=\footnotesize] {$p(T \mid D)$} (T);
    \node[above=0.1cm of D, font=\footnotesize] {$p(D)$};
  \end{tikzpicture}
\end{center}
The problem gives $p(D)$ and $p(T \mid D)$ for all values. Two rules give
the other probabilities:
\begin{align*}
  &\text{R1. Against the arrow (Bayes' theorem):}
    && p(D \mid T) = \frac{p(T \mid D) \, p(D)}{p(T)}, \\
  &\text{R2. Marginal probability (chain rule):}
    && p(T) = \sum_{d} p(T \mid D = d) \, p(D = d).
\end{align*}

\solpart{Solution}
\emph{Computation graph.} The goal is at the top. Each arrow shows the rule
and the formula for the node above it. Shaded nodes are given.
\begin{center}
  \begin{tikzpicture}[cg]
    \node[cg goal]  (G)  at (2, 0)    {$p(D \mid T)$};
    \node[cg node]  (T)  at (2, -2)   {$p(T)$};
    \node[cg given] (TD) at (0, -4)   {$p(T \mid D)$};
    \node[cg given] (D)  at (7, -4)   {$p(D)$};
    \coordinate (g) at (2, -1);
    \draw (T) -- (g);
    \draw (TD) to[out=90, in=180] (g);
    \draw (D) to[out=90, in=0] (g);
    \draw[->] (g) -- node[cg op, right]
      {R1: $p(T \mid D) \, p(D) \,/\, p(T)$} (G);
    \coordinate (t) at (2, -3);
    \draw (TD) -- (t);
    \draw (D) -- (t);
    \draw[->] (t) -- node[cg op, right]
      {R2: $\sum_d p(T \mid D{=}d) \, p(D{=}d)$} (T);
  \end{tikzpicture}
\end{center}

\solstep{Step 1. Find $p(T = 1)$ with R2.}
\begin{align*}
  p(T = 1)
  &= p(T = 1 \mid D = 1) \, p(D = 1) + p(T = 1 \mid D = 0) \, p(D = 0) \\
  &= 0.99 \cdot 0.0001 + 0.01 \cdot 0.9999 \\
  &= 0.000099 + 0.009999
   = 0.010098.
\end{align*}

\solstep{Step 2. Find $p(D = 1 \mid T = 1)$ with R1.}
\begin{align*}
  p(D = 1 \mid T = 1)
  &= \frac{p(T = 1 \mid D = 1) \, p(D = 1)}{p(T = 1)}
   = \frac{0.99 \cdot 0.0001}{0.010098}
   = \frac{0.000099}{0.010098} \\
  &= \frac{99}{10098}
    && \text{(multiply by $10^6$)} \\
  &= \frac{99}{99 \cdot 102}
   = \frac{1}{102}
   \approx 0.0098.
\end{align*}

\solpart{Conclusion}
The probability increases from 0.01\% to only 1\%. In 10\,000 people, the
test finds $10\,000 \cdot 0.000099 \approx 1$ sick person and
$10\,000 \cdot 0.009999 \approx 100$ healthy people. Thus most positive
results of a mass test for a rare disease are false. A second positive
independent test uses $p(D = 1 \mid T_1 = 1) = 1/102$ as the prior:
\begin{align*}
  p(D = 1 \mid T_1 = 1, T_2 = 1)
  &= \frac{\frac{1}{102} \cdot 0.99}
          {\frac{1}{102} \cdot 0.99 + \bigl(1 - \frac{1}{102}\bigr) \cdot 0.01} \\
  &= \frac{\frac{1}{102} \cdot 0.99}
          {\frac{1}{102} \cdot 0.99 + \frac{101}{102} \cdot 0.01} \\
  &= \frac{0.99}{0.99 + 101 \cdot 0.01}
    && \text{(multiply by 102)} \\
  &= \frac{0.99}{0.99 + 1.01}
   = \frac{0.99}{2}
   = 0.495.
\end{align*}
\end{solution}
