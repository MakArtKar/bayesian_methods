\label{prob:medical-test}
A medical test finds a serious disease in a patient. The test is 99\%
accurate: if the patient has the disease, the test is positive with
probability 99\%; if the patient does not have the disease, the test is
negative with probability 99\%. The disease is rare: only one person in
10\,000 has it. Find the probability that the patient really has the
disease.

\begin{solution}
\solpart{Interpretation}
$D = 1$: the patient has the disease. $T = 1$: the test result is positive.
\[
  p(D = 1) = 10^{-4}, \qquad
  p(T = 1 \mid D = 1) = 0.99, \qquad
  p(T = 0 \mid D = 0) = 0.99.
\]
Find $p(D = 1 \mid T = 1)$.

\solpart{Idea}
The disease causes the test result: the graph is $D \to T$. The source node
and the edge give the joint distribution:
\[
  p(D, T) = p(D) \, p(T \mid D).
\]
The joint distribution gives any other probability with two operations:
\begin{align*}
  p(T) &= \sum_{D} p(D, T)
    && \text{(marginal; for continuous $x$: $p(y) = \textstyle\int p(x, y) \, dx$)}, \\
  p(D \mid T) &= \frac{p(D, T)}{p(T)}
    && \text{(conditional)}.
\end{align*}
Together, they give Bayes' theorem:
\[
  p(D \mid T)
  = \frac{p(D, T)}{p(T)}
  = \frac{p(D, T)}{\sum_{D'} p(D', T)}
  = \frac{p(D) \, p(T \mid D)}{\sum_{D'} p(D') \, p(T \mid D')}.
\]

\solpart{Solution}
\solstep{Step 1. Write the goal.}
\[
  p(D = 1 \mid T = 1) = \frac{p(D = 1, T = 1)}{p(T = 1)}.
\]

\solstep{Step 2. Find the numerator $p(D = 1, T = 1)$.}
\[
  p(D = 1, T = 1) = p(D = 1) \, p(T = 1 \mid D = 1)
  = 0.0001 \cdot 0.99 = 0.000099.
\]

\solstep{Step 3. Find the denominator $p(T = 1)$.}
\[
  p(T = 1)
  = p(D = 1, T = 1) + p(D = 0, T = 1)
  = 0.000099 + p(D = 0, T = 1)
  \qquad \text{(Step 2)}.
\]

\solstep{Step 3a. Find $p(D = 0, T = 1)$ for Step 3.}
\begin{align*}
  p(D = 0, T = 1) &= p(D = 0) \, p(T = 1 \mid D = 0), \\
  p(D = 0) &= 1 - p(D = 1) = 1 - 0.0001 = 0.9999, \\
  p(T = 1 \mid D = 0) &= 1 - p(T = 0 \mid D = 0) = 1 - 0.99 = 0.01
    && \text{(false positive)}, \\
  p(D = 0, T = 1) &= 0.9999 \cdot 0.01 = 0.009999.
\end{align*}

\solstep{Step 3b. Substitute Step 3a into Step 3.}
\[
  p(T = 1) = 0.000099 + 0.009999 = 0.010098.
\]

\solstep{Step 4. Substitute Steps 2 and 3 into Step 1.}
\[
  p(D = 1 \mid T = 1)
  = \frac{0.000099}{0.010098}
  = \frac{99}{99 \cdot 102}
  = \frac{1}{102}
  \approx 0.0098.
\]

\solpart{Conclusion}
The probability increases from 0.01\% to only 1\%. In 10\,000 people, the
test finds $10\,000 \cdot 0.000099 \approx 1$ sick person and
$10\,000 \cdot 0.009999 \approx 100$ healthy people. Thus most positive
results of a mass test for a rare disease are false. A second positive
independent test uses $1/102$ as the prior:
\[
  p(D = 1 \mid T_1 = 1, T_2 = 1)
  = \frac{\frac{1}{102} \cdot 0.99}
         {\frac{1}{102} \cdot 0.99 + \frac{101}{102} \cdot 0.01}
  = \frac{0.99}{0.99 + 1.01}
  = 0.495.
\]
\end{solution}
