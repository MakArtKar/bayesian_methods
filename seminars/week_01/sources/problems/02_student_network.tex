\label{prob:student-network}
Consider the following probabilistic model. At a seminar, a student answers
badly ($A = 1$) or well ($A = 0$). This depends on whether the student has
depression ($D = 1$ or $D = 0$) and whether the student went to a party the
night before ($P = 1$ or $P = 0$). The party can also give the student a
headache ($H = 1$). A bad answer makes the teacher unhappy ($T = 1$).
Figure~\ref{fig:student-network} shows the causal relations in the model.
The probabilities are:
\begin{center}
  \begin{tabular}[t]{c|cc}
    \toprule
    $p(A = 1 \mid P, D)$ & $P$ & $D$ \\
    \midrule
    0.999 & 1 & 1 \\
    0.9   & 1 & 0 \\
    0.9   & 0 & 1 \\
    0.01  & 0 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(H = 1 \mid P)$ & $P$ \\
    \midrule
    0.9 & 1 \\
    0.2 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(T = 1 \mid A)$ & $A$ \\
    \midrule
    0.95 & 1 \\
    0.5  & 0 \\
    \bottomrule
  \end{tabular}

  \medskip
  $p(P = 1) = 0.2, \qquad p(D = 1) = 0.4.$
\end{center}
Find $p(P = 1 \mid H = 1)$, $p(P = 1 \mid T = 1)$, and
$p(P = 1 \mid T = 1, H = 1)$.

  \centering

\begin{solution}
\solpart{Idea}
\emph{Main question:} how to find $p(G_x \mid G_y)$ for any groups of
variables $G_x$ and $G_y$? Problem~\ref{prob:medical-test} is a special
case.

\emph{Step 1. Reduce the question to joint distributions.}
\[
  p(G_x \mid G_y) = \frac{p(G_x, G_y)}{p(G_y)}.
\]
Thus it is sufficient to find the joint distribution $p(G)$ of any group
$G$.

\emph{Step 2. Find the factors for $p(G)$.} The problem gives the
distribution of each source and the distribution of each other node given
its parents. The chain rule gives any joint distribution from these
factors. Example for $G = \{H, A\}$ in the graph of this problem. $H$ and
$A$ depend on $P$ and $D$, so add $P$ and $D$:
\begin{align*}
  p(P, D, A, H)
  &= p(P, D, A) \, p(H \mid P, D, A)
    && \text{(chain rule)} \\
  &= p(P, D, A) \, p(H \mid P)
    && \text{(parent of $H$: $P$)} \\
  &= p(P, D) \, p(A \mid P, D) \, p(H \mid P)
    && \text{(chain rule; parents of $A$: $P$, $D$)} \\
  &= p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P)
    && \text{(chain rule; $D$ has no parents)}.
\end{align*}
The factor $p(T \mid A)$ is not in the product. The product contains the
factors of the sources and the arrows from which a path goes to $G$, and
only these factors. From $T$, no path goes to $H$ or $A$:
\begin{center}
  \crossgraph{k}{k}{g}{g}{x}
\end{center}

\emph{Trick:} write the joint distribution of all variables one time. For
each $G$, strike out the factors of the nodes from which no path goes to
$G$:
\[
  p(P, D, A, H, T)
  = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, \strike{p(T \mid A)}.
\]

\emph{Step 3. Sum over the other variables.} Sum the product of the kept
factors over the kept variables that are not in $G$:
\[
  p(H, A)
  = \sum_{p} \sum_{d} p(P = p) \, p(D = d) \, p(A \mid P = p, D = d) \,
    p(H \mid P = p).
\]
Move the factors that do not depend on a summation variable out of its
sum, and calculate each factor.

\solpart{Solution}
The tables give $p(X = 1 \mid \text{parents})$ for each node $X$, and
$p(X = 0 \mid \text{parents}) = 1 - p(X = 1 \mid \text{parents})$. In each
graph, the nodes of $G$ are blue. A red cross marks a struck-out
factor: on a source, or on the arrows into a node.

\emph{Joint distribution.} Apply the chain rule one variable at a time,
from the children to the parents. A missing arrow means no direct
influence: given its parents, a node does not depend on the other nodes in
the condition.
\begin{align*}
  p(P, D, A, H, T)
  &= p(P, D, A, H) \, p(T \mid P, D, A, H)
   && \text{(chain rule)} \\
  &= p(P, D, A, H) \, p(T \mid A)
   && \text{(parent of $T$: $A$)} \\
  p(P, D, A, H)
  &= p(P, D, A) \, p(H \mid P, D, A)
   && \text{(chain rule)} \\
  &= p(P, D, A) \, p(H \mid P)
   && \text{(parent of $H$: $P$)} \\
  p(P, D, A)
  &= p(P, D) \, p(A \mid P, D)
   && \text{(chain rule; parents of $A$: $P$, $D$)} \\
  p(P, D)
  &= p(P) \, p(D \mid P)
   && \text{(chain rule)} \\
  &= p(P) \, p(D)
   && \text{($D$ has no parents)}
\end{align*}
Thus
\[
  p(P, D, A, H, T)
  = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, p(T \mid A).
\]

\solsubproblem{(a) Find $p(P = 1 \mid H = 1)$.}

\solstep{Step 1. Write the goal as a ratio (Idea, Step~1).}
\[
  p(P = 1 \mid H = 1) = \frac{p(P = 1, H = 1)}{p(H = 1)}.
\]

\solstep{Step 2. Strike out the factors for the numerator (Idea, Step~2): $G = \{P, H\}$.}
\begin{center}
  \crossgraph{g}{x}{g}{x}{x}
\end{center}
\[
  p(P, D, A, H, T)
  = p(P) \, \strike{p(D)} \, \strike{p(A \mid P, D)} \, p(H \mid P) \, \strike{p(T \mid A)}.
\]
Keep the other factors:
\[
  p(P, H) = p(P) \, p(H \mid P).
\]

\solstep{Step 3. Find the numerator (Idea, Step~3).}
No other variables are kept, so there is no sum:
\[
  p(P = 1, H = 1)
  = p(P = 1) \, p(H = 1 \mid P = 1)
  = 0.2 \cdot 0.9
  = 0.18.
\]

\solstep{Step 4. Strike out the factors for the denominator (Idea, Step~2): $G = \{H\}$.}
\begin{center}
  \crossgraph{k}{x}{g}{x}{x}
\end{center}
\[
  p(P, D, A, H, T)
  = p(P) \, \strike{p(D)} \, \strike{p(A \mid P, D)} \, p(H \mid P) \, \strike{p(T \mid A)}.
\]
Keep the other factors:
\[
  p(P, H) = p(P) \, p(H \mid P).
\]

\solstep{Step 5. Find the denominator (Idea, Step~3).}
\begin{align*}
  p(H = 1)
  &= \sum_{p} p(P = p) \, p(H = 1 \mid P = p)
    && \text{(Step 4)} \\
  &= 0.2 \cdot 0.9 + (1 - 0.2) \cdot 0.2
   = 0.18 + 0.16
   = 0.34.
\end{align*}

\solstep{Step 6. Divide.}
\[
  p(P = 1 \mid H = 1)
  = \frac{0.18}{0.34}
  = \frac{9}{17}
  \approx 0.529.
\]

\solsubproblem{(b) Find $p(P = 1 \mid T = 1)$.}

\solstep{Step 1. Write the goal as a ratio (Idea, Step~1).}
\[
  p(P = 1 \mid T = 1) = \frac{p(P = 1, T = 1)}{p(T = 1)}.
\]

\solstep{Step 2. Strike out the factors for the numerator (Idea, Step~2): $G = \{P, T\}$.}
\begin{center}
  \crossgraph{g}{k}{x}{k}{g}
\end{center}
\[
  p(P, D, A, H, T)
  = p(P) \, p(D) \, p(A \mid P, D) \, \strike{p(H \mid P)} \, p(T \mid A).
\]
Keep the other factors:
\[
  p(P, D, A, T) = p(P) \, p(D) \, p(A \mid P, D) \, p(T \mid A).
\]

\solstep{Step 3. Find the numerator (Idea, Step~3).}
\begin{align*}
  &p(P = 1, T = 1) \\
  &= \sum_{a} \sum_{d} p(P = 1) \, p(D = d) \, p(A = a \mid P = 1, D = d) \,
     p(T = 1 \mid A = a)
    && \text{(Step 2)} \\
  &= p(P = 1) \sum_{a} p(T = 1 \mid A = a) \sum_{d}
     p(A = a \mid P = 1, D = d) \, p(D = d)
    && \text{(constants out)} \\
  &= 0.2 \, \bigl[ 0.95 \, \bigl(0.999 \cdot 0.4 + 0.9 \cdot (1 - 0.4)\bigr)
    && \text{($a = 1$, $a = 0$)} \\
  &\qquad + 0.5 \, \bigl((1 - 0.999) \cdot 0.4 + (1 - 0.9) \cdot (1 - 0.4)\bigr)
     \bigr] \\
  &= 0.2 \, [0.95 \cdot 0.9396 + 0.5 \cdot 0.0604] \\
  &= 0.2 \, [0.89262 + 0.0302]
   = 0.2 \cdot 0.92282
   = 0.184564.
\end{align*}

\solstep{Step 4. Strike out the factors for the denominator (Idea, Step~2): $G = \{T\}$.}
\begin{center}
  \crossgraph{k}{k}{x}{k}{g}
\end{center}
\[
  p(P, D, A, H, T)
  = p(P) \, p(D) \, p(A \mid P, D) \, \strike{p(H \mid P)} \, p(T \mid A).
\]
Keep the other factors:
\[
  p(P, D, A, T) = p(P) \, p(D) \, p(A \mid P, D) \, p(T \mid A).
\]

\solstep{Step 5. Find the denominator (Idea, Step~3).}
\begin{align*}
  &p(T = 1) \\
  &= \sum_{p} \sum_{a} \sum_{d} p(P = p) \, p(D = d) \,
     p(A = a \mid P = p, D = d) \, p(T = 1 \mid A = a)
    && \text{(Step 4)} \\
  &= \sum_{p} p(P = p) \sum_{a} p(T = 1 \mid A = a) \sum_{d}
     p(A = a \mid P = p, D = d) \, p(D = d)
    && \text{(constants out)} \\
  &= 0.184564 + (1 - 0.2) \, \bigl[ 0.95 \, (0.9 \cdot 0.4 + 0.01 \cdot 0.6)
    && \text{($p = 1$: Step 3)} \\
  &\qquad + 0.5 \, \bigl((1 - 0.9) \cdot 0.4 + (1 - 0.01) \cdot 0.6\bigr) \bigr] \\
  &= 0.184564 + 0.8 \, [0.95 \cdot 0.366 + 0.5 \cdot 0.634] \\
  &= 0.184564 + 0.8 \, [0.3477 + 0.317]
   = 0.184564 + 0.8 \cdot 0.6647 \\
  &= 0.184564 + 0.53176
   = 0.716324.
\end{align*}

\solstep{Step 6. Divide.}
\[
  p(P = 1 \mid T = 1)
  = \frac{0.184564}{0.716324}
  \approx 0.258.
\]

\solsubproblem{(c) Find $p(P = 1 \mid T = 1, H = 1)$.}

\solstep{Step 1. Write the goal as a ratio (Idea, Step~1).}
\[
  p(P = 1 \mid T = 1, H = 1)
  = \frac{p(P = 1, H = 1, T = 1)}{p(H = 1, T = 1)}.
\]

\solstep{Step 2. Strike out the factors for the numerator (Idea, Step~2): $G = \{P, H, T\}$.}
\begin{center}
  \crossgraph{g}{k}{g}{k}{g}
\end{center}
\[
  p(P, D, A, H, T)
  = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, p(T \mid A).
\]
Keep the other factors:
\[
  p(P, D, A, H, T) = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, p(T \mid A).
\]

\solstep{Step 3. Find the numerator (Idea, Step~3).}
\begin{align*}
  &p(P = 1, H = 1, T = 1) \\
  &= \sum_{a} \sum_{d} p(P = 1) \, p(D = d) \, p(A = a \mid P = 1, D = d)
    && \text{(Step 2)} \\
  &\qquad \cdot p(H = 1 \mid P = 1) \, p(T = 1 \mid A = a) \\
  &= p(H = 1 \mid P = 1)
    && \text{(constants out)} \\
  &\qquad \cdot p(P = 1) \sum_{a} p(T = 1 \mid A = a)
     \sum_{d} p(A = a \mid P = 1, D = d) \, p(D = d) \\
  &= 0.9 \cdot 0.184564
    && \text{((b), Step 3)} \\
  &= 0.1661076.
\end{align*}

\solstep{Step 4. Strike out the factors for the denominator (Idea, Step~2): $G = \{H, T\}$.}
\begin{center}
  \crossgraph{k}{k}{g}{k}{g}
\end{center}
\[
  p(P, D, A, H, T)
  = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, p(T \mid A).
\]
Keep the other factors:
\[
  p(P, D, A, H, T) = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, p(T \mid A).
\]

\solstep{Step 5. Find the denominator (Idea, Step~3).}
\begin{align*}
  &p(H = 1, T = 1) \\
  &= \sum_{p} \sum_{a} \sum_{d} p(P = p) \, p(D = d) \,
     p(A = a \mid P = p, D = d)
    && \text{(Step 4)} \\
  &\qquad \cdot p(H = 1 \mid P = p) \, p(T = 1 \mid A = a) \\
  &= \sum_{p} p(H = 1 \mid P = p)
    && \text{(constants out)} \\
  &\qquad \cdot p(P = p) \sum_{a} p(T = 1 \mid A = a)
     \sum_{d} p(A = a \mid P = p, D = d) \, p(D = d) \\
  &= 0.9 \cdot 0.184564 + 0.2 \cdot 0.53176
    && \text{((b), Step 5)} \\
  &= 0.1661076 + 0.106352
   = 0.2724596.
\end{align*}

\solstep{Step 6. Divide.}
\[
  p(P = 1 \mid T = 1, H = 1)
  = \frac{0.1661076}{0.2724596}
  \approx 0.610.
\]

\solpart{Conclusion}
\[
  \underbrace{0.2}_{\text{prior}}
  \;\to\;
  \underbrace{0.529}_{H = 1}, \quad
  \underbrace{0.258}_{T = 1}, \quad
  \underbrace{0.610}_{H = 1,\, T = 1}.
\]
An unhappy teacher is weak evidence: without the party, the teacher is
also often unhappy ($0.6647$ in Step~5 of (b)). Two observations together give the strongest evidence.
\end{solution}
