\label{prob:student-network}
Consider the following probabilistic model. At a seminar, a student answers
badly ($A = 1$) or well ($A = 0$). This depends on whether the student has
depression ($D = 1$ or $D = 0$) and whether the student went to a party the
night before ($P = 1$ or $P = 0$). The party can also give the student a
headache ($H = 1$). A bad answer makes the teacher unhappy ($T = 1$).
Figure~\ref{fig:student-network} shows the causal relations in the model.
The probabilities are:
\begin{center}
  \begin{tabular}[t]{c|cc}
    \toprule
    $p(A = 1 \mid P, D)$ & $P$ & $D$ \\
    \midrule
    0.999 & 1 & 1 \\
    0.9   & 1 & 0 \\
    0.9   & 0 & 1 \\
    0.01  & 0 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(H = 1 \mid P)$ & $P$ \\
    \midrule
    0.9 & 1 \\
    0.2 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(T = 1 \mid A)$ & $A$ \\
    \midrule
    0.95 & 1 \\
    0.5  & 0 \\
    \bottomrule
  \end{tabular}

  \medskip
  $p(P = 1) = 0.2, \qquad p(D = 1) = 0.4.$
\end{center}
Find $p(P = 1 \mid H = 1)$, $p(P = 1 \mid T = 1)$, and
$p(P = 1 \mid T = 1, H = 1)$.

  \centering

      node distance=0.3cm and 0.5cm,

      >={Stealth},

    \node[var] (P) {$P$};
    \node[var, right=2cm of P] (D) {$D$};
    \node[var, below left=of P] (H) {$H$};
    \node[var, below right=of P] (A) {$A$};
    \node[var, below=of A] (T) {$T$};
    \draw[->] (P) -- (H);
    \draw[->] (P) -- (A);

  \end{tikzpicture}
  \caption{Causal relations for Problem~\ref{prob:student-network}}
  \label{fig:student-network}
\end{figure}

\begin{solution}
\solpart{Idea}
Use the plan of Problem~\ref{prob:medical-test} on a larger graph. The
tables give $p(X = 1 \mid \text{parents})$ for each node $X$, and
$p(X = 0 \mid \text{parents}) = 1 - p(X = 1 \mid \text{parents})$.

\emph{Joint distribution.} Apply the chain rule one variable at a time,
from the children to the parents. A missing arrow means no direct
influence: given its parents, a node does not depend on the other nodes in
the condition.
\begin{align*}
  p(P, D, A, H, T)
  &= p(P, D, A, H) \, p(T \mid P, D, A, H) \\
  &= p(P, D, A, H) \, p(T \mid A)
   && \text{(parent of $T$: $A$)} \\
  p(P, D, A, H)
  &= p(P, D, A) \, p(H \mid P, D, A) \\
  &= p(P, D, A) \, p(H \mid P)
   && \text{(parent of $H$: $P$)} \\
  p(P, D, A)
  &= p(P, D) \, p(A \mid P, D)
   && \text{(parents of $A$: $P$, $D$)} \\
  p(P, D)
  &= p(P) \, p(D \mid P) \\
  &= p(P) \, p(D)
   && \text{($D$ has no parents)}
\end{align*}
Thus
\[
  p(P, D, A, H, T)
  = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, p(T \mid A).
\]

\solpart{Solution}
\solsubproblem{(a) Find $p(P = 1 \mid H = 1)$.}

\solstep{Step 1. Write the goal as a ratio.}
\[
  p(P = 1 \mid H = 1) = \frac{p(P = 1, H = 1)}{p(H = 1)}.
\]

\solstep{Step 2. Find the numerator.}
Sum the joint distribution over $D$, $A$, and $T$. Use any value $p$ of
$P$, because Step~3 also needs $P = 0$:
\begin{align*}
  &p(P = p, H = 1) \\
  &= \sum_{d} \sum_{a} \sum_{t} p(P = p) \, p(D = d) \,
     p(A = a \mid P = p, D = d) \\
  &\qquad \cdot p(H = 1 \mid P = p) \, p(T = t \mid A = a) \\
  &= p(P = p) \, p(H = 1 \mid P = p) \\
  &\qquad \cdot \sum_{d} p(D = d) \sum_{a} p(A = a \mid P = p, D = d)
     \sum_{t} p(T = t \mid A = a)
    && \text{(constants out)} \\
  &= p(P = p) \, p(H = 1 \mid P = p)
     \sum_{d} p(D = d) \sum_{a} p(A = a \mid P = p, D = d)
    && \textstyle\text{($\sum_t = 1$)} \\
  &= p(P = p) \, p(H = 1 \mid P = p) \sum_{d} p(D = d)
    && \textstyle\text{($\sum_a = 1$)} \\
  &= p(P = p) \, p(H = 1 \mid P = p)
    && \textstyle\text{($\sum_d = 1$)}.
\end{align*}
Substitute the values:
\begin{align*}
  p(P = 1, H = 1) &= 0.2 \cdot 0.9 = 0.18, \\
  p(P = 0, H = 1) &= 0.8 \cdot 0.2 = 0.16.
\end{align*}

\solstep{Step 3. Find the denominator.}
\[
  p(H = 1)
  = \sum_{p} p(P = p, H = 1)
  = 0.18 + 0.16
  = 0.34.
\]

\solstep{Step 4. Divide.}
\[
  p(P = 1 \mid H = 1)
  = \frac{0.18}{0.34}
  = \frac{9}{17}
  \approx 0.529.
\]

\solsubproblem{(b) Find $p(P = 1 \mid T = 1)$.}

\solstep{Step 1. Write the goal as a ratio.}
\[
  p(P = 1 \mid T = 1) = \frac{p(P = 1, T = 1)}{p(T = 1)}.
\]

\solstep{Step 2. Sum the joint distribution over $H$, $A$, and $D$.}
\begin{align*}
  &p(P = p, T = 1) \\
  &= \sum_{d} \sum_{a} \sum_{h} p(P = p) \, p(D = d) \,
     p(A = a \mid P = p, D = d) \\
  &\qquad \cdot p(H = h \mid P = p) \, p(T = 1 \mid A = a) \\
  &= p(P = p) \sum_{d} p(D = d) \sum_{a} p(A = a \mid P = p, D = d) \,
     p(T = 1 \mid A = a) \\
  &\qquad \cdot \sum_{h} p(H = h \mid P = p)
    && \text{(constants out)} \\
  &= p(P = p) \sum_{d} p(D = d) \sum_{a} p(A = a \mid P = p, D = d) \,
     p(T = 1 \mid A = a)
    && \textstyle\text{($\sum_h = 1$)}.
\end{align*}

\solstep{Step 3. Calculate the sum over $A$.}
Let $q = p(A = 1 \mid P = p, D = d)$. Then $p(A = 0 \mid P = p, D = d) = 1 - q$:
\[
  \sum_{a} p(A = a \mid P = p, D = d) \, p(T = 1 \mid A = a)
  = 0.95 \, q + 0.5 \, (1 - q)
  = 0.5 + 0.45 \, q.
\]

\solstep{Step 4. Calculate the sum over $D$.}
Substitute Step~3 and move the constants out:
\begin{align*}
  &\sum_{d} p(D = d) \bigl(0.5 + 0.45 \, p(A = 1 \mid P = p, D = d)\bigr) \\
  &= 0.5 \sum_{d} p(D = d)
   + 0.45 \sum_{d} p(D = d) \, p(A = 1 \mid P = p, D = d) \\
  &= 0.5 + 0.45 \sum_{d} p(D = d) \, p(A = 1 \mid P = p, D = d)
    && \textstyle\text{($\sum_d p(D = d) = 1$)}.
\end{align*}
Let $g(p)$ be this value. Substitute the tables:
\begin{align*}
  g(1) &= 0.5 + 0.45 \, (0.4 \cdot 0.999 + 0.6 \cdot 0.9)
        = 0.5 + 0.45 \cdot 0.9396
        = 0.5 + 0.42282
        = 0.92282, \\
  g(0) &= 0.5 + 0.45 \, (0.4 \cdot 0.9 + 0.6 \cdot 0.01)
        = 0.5 + 0.45 \cdot 0.366
        = 0.5 + 0.1647
        = 0.6647.
\end{align*}

\solstep{Step 5. Find the numerator.}
From Steps 2 and 4, $p(P = p, T = 1) = p(P = p) \, g(p)$:
\begin{align*}
  p(P = 1, T = 1) &= 0.2 \cdot 0.92282 = 0.184564, \\
  p(P = 0, T = 1) &= 0.8 \cdot 0.6647 = 0.53176.
\end{align*}

\solstep{Step 6. Find the denominator.}
\[
  p(T = 1)
  = \sum_{p} p(P = p, T = 1)
  = 0.184564 + 0.53176
  = 0.716324.
\]

\solstep{Step 7. Divide.}
\[
  p(P = 1 \mid T = 1)
  = \frac{0.184564}{0.716324}
  \approx 0.258.
\]

\solsubproblem{(c) Find $p(P = 1 \mid T = 1, H = 1)$.}

\solstep{Step 1. Write the goal as a ratio.}
\[
  p(P = 1 \mid T = 1, H = 1)
  = \frac{p(P = 1, H = 1, T = 1)}{p(H = 1, T = 1)}.
\]

\solstep{Step 2. Find the numerator.}
Sum the joint distribution over $D$ and $A$. The remaining sum is $g(p)$
from Step~4 of (b):
\begin{align*}
  &p(P = p, H = 1, T = 1) \\
  &= \sum_{d} \sum_{a} p(P = p) \, p(D = d) \,
     p(A = a \mid P = p, D = d) \\
  &\qquad \cdot p(H = 1 \mid P = p) \, p(T = 1 \mid A = a) \\
  &= p(P = p) \, p(H = 1 \mid P = p) \\
  &\qquad \cdot \sum_{d} p(D = d) \sum_{a} p(A = a \mid P = p, D = d) \,
     p(T = 1 \mid A = a)
    && \text{(constants out)} \\
  &= p(P = p) \, p(H = 1 \mid P = p) \, g(p).
\end{align*}
Substitute the values:
\begin{align*}
  p(P = 1, H = 1, T = 1) &= 0.2 \cdot 0.9 \cdot 0.92282
    = 0.18 \cdot 0.92282 = 0.1661076, \\
  p(P = 0, H = 1, T = 1) &= 0.8 \cdot 0.2 \cdot 0.6647
    = 0.16 \cdot 0.6647 = 0.106352.
\end{align*}

\solstep{Step 3. Find the denominator.}
\[
  p(H = 1, T = 1)
  = \sum_{p} p(P = p, H = 1, T = 1)
  = 0.1661076 + 0.106352
  = 0.2724596.
\]

\solstep{Step 4. Divide.}
\[
  p(P = 1 \mid T = 1, H = 1)
  = \frac{0.1661076}{0.2724596}
  \approx 0.610.
\]

\solpart{Conclusion}
\[
  \underbrace{0.2}_{\text{prior}}
  \;\to\;
  \underbrace{0.529}_{H = 1}, \quad
  \underbrace{0.258}_{T = 1}, \quad
  \underbrace{0.610}_{H = 1,\, T = 1}.
\]
An unhappy teacher is weak evidence: $p(T = 1 \mid P = 0) = 0.6647$ is
also high. Two observations together give the strongest evidence.
\end{solution}
