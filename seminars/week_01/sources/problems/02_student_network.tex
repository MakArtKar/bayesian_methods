\label{prob:student-network}
Consider the following probabilistic model. At a seminar, a student answers
badly ($A = 1$) or well ($A = 0$). This depends on whether the student has
depression ($D = 1$ or $D = 0$) and whether the student went to a party the
night before ($P = 1$ or $P = 0$). The party can also give the student a
headache ($H = 1$). A bad answer makes the teacher unhappy ($T = 1$).
Figure~\ref{fig:student-network} shows the causal relations in the model.
The probabilities are:
\begin{center}
  \begin{tabular}[t]{c|cc}
    \toprule
    $p(A = 1 \mid P, D)$ & $P$ & $D$ \\
    \midrule
    0.999 & 1 & 1 \\
    0.9   & 1 & 0 \\
    0.9   & 0 & 1 \\
    0.01  & 0 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(H = 1 \mid P)$ & $P$ \\
    \midrule
    0.9 & 1 \\
    0.2 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(T = 1 \mid A)$ & $A$ \\
    \midrule
    0.95 & 1 \\
    0.5  & 0 \\
    \bottomrule
  \end{tabular}

  \medskip
  $p(P = 1) = 0.2, \qquad p(D = 1) = 0.4.$
\end{center}
Find $p(P = 1 \mid H = 1)$, $p(P = 1 \mid T = 1)$, and
$p(P = 1 \mid T = 1, H = 1)$.

\begin{solution}
\solpart{Idea}
\emph{Main question:} how to find $p(G_x \mid G_y)$ for any groups of
variables $G_x$ and $G_y$? Problem~\ref{prob:medical-test} is a special
case.

\emph{Step 1. Reduce the question to joint distributions.}
\[
  p(G_x \mid G_y) = \frac{p(G_x, G_y)}{p(G_y)}.
\]
Thus it is sufficient to find the joint distribution $p(G)$ of any group
$G$.

\emph{Step 2. Find $p(G, \Ganc)$ with the chain rule.} The problem gives the
distribution of each source node and the distribution of each other node
given its parents. The chain rule gives any joint distribution from these
factors. Example for the graph $X \to Z \leftarrow Y$:
\begin{align*}
  p(X, Y, Z)
  &= p(X, Y) \, p(Z \mid X, Y)
    && \text{(chain rule)} \\
  &= p(X) \, p(Y \mid X) \, p(Z \mid X, Y)
    && \text{(chain rule)} \\
  &= p(X) \, p(Y) \, p(Z \mid X, Y)
    && \text{($Y$ has no parents)}.
\end{align*}
To find $p(G)$, only the ancestors of $G$ are necessary. Let $\Ganc$ be the
ancestors of $G$ that are not in $G$, and $\Grest$ all other variables.

\emph{Trick:} write the joint distribution of all variables one time. Then
strike out each factor with a variable of $\Grest$. The other factors give
$p(G, \Ganc)$. No arrow goes from $\Grest$ to $G$ or $\Ganc$, so these
factors do not contain $\Grest$:
\begin{center}
  \begin{tikzpicture}[
      >={Stealth},
      dot/.style={circle, draw, fill=white, minimum size=0.3cm, inner sep=0pt},
      box/.style={draw, rounded corners, inner sep=7pt},
      meta/.style={->, very thick},
    ]
    \node[dot] (a1) at (0, 0) {};
    \node[dot] (a2) at (1, -0.5) {};
    \node[dot] (a3) at (2, 0) {};
    \draw[->] (a1) -- (a2);
    \draw[->] (a3) -- (a2);
    \node[dot] (g1) at (-2, -2.4) {};
    \node[dot] (g2) at (-1, -2.9) {};
    \node[dot] (g3) at (-2, -3.4) {};
    \draw[->] (g1) -- (g2);
    \draw[->] (g3) -- (g2);
    \node[dot] (r1) at (3, -2.4) {};
    \node[dot] (r2) at (4, -2.9) {};
    \node[dot] (r3) at (3, -3.4) {};
    \draw[->] (r1) -- (r2);
    \draw[->] (r2) -- (r3);
    \begin{scope}[on background layer]
      \node[box, anc, fit=(a1) (a2) (a3)] (BA) {};
      \node[box, grp, fit=(g1) (g2) (g3)] (BG) {};
      \node[box, rst, fit=(r1) (r2) (r3)] (BR) {};
    \end{scope}
    \node[above=0.05cm of BA] {$\Ganc$};
    \node[left=0.05cm of BG] {$G$};
    \node[right=0.05cm of BR] {$\Grest$};
    \draw[meta] (BA) -- (BG);
    \draw[meta] (BA) -- (BR);
    \draw[meta] (BG) -- (BR);
  \end{tikzpicture}
\end{center}

\emph{Step 3. Sum over $\Ganc$.}
\[
  p(G) = \sum_{\Ganc} p(G, \Ganc).
\]
Write the product from Step~2 under the sum, move the factors that do not
depend on a summation variable out of its sum, and calculate each factor.

\solpart{Solution}
The tables give $p(X = 1 \mid \text{parents})$ for each node $X$, and
$p(X = 0 \mid \text{parents}) = 1 - p(X = 1 \mid \text{parents})$.

\emph{Joint distribution.} Apply the chain rule one variable at a time,
from the children to the parents. A missing arrow means no direct
influence: given its parents, a node does not depend on the other nodes in
the condition.
\begin{align*}
  p(P, D, A, H, T)
  &= p(P, D, A, H) \, p(T \mid P, D, A, H)
   && \text{(chain rule)} \\
  &= p(P, D, A, H) \, p(T \mid A)
   && \text{(parent of $T$: $A$)} \\
  p(P, D, A, H)
  &= p(P, D, A) \, p(H \mid P, D, A)
   && \text{(chain rule)} \\
  &= p(P, D, A) \, p(H \mid P)
   && \text{(parent of $H$: $P$)} \\
  p(P, D, A)
  &= p(P, D) \, p(A \mid P, D)
   && \text{(chain rule; parents of $A$: $P$, $D$)} \\
  p(P, D)
  &= p(P) \, p(D \mid P)
   && \text{(chain rule)} \\
  &= p(P) \, p(D)
   && \text{($D$ has no parents)}
\end{align*}
Thus
\[
  p(P, D, A, H, T)
  = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, p(T \mid A).
\]

\solsubproblem{(a) Find $p(P = 1 \mid H = 1)$.}

\solstep{Step 1. Write the goal as a ratio (Idea, Step~1).}
\[
  p(P = 1 \mid H = 1) = \frac{p(P = 1, H = 1)}{p(H = 1)}.
\]

\solstep{Step 2. Find $p(G, \Ganc)$ for the numerator (Idea, Step~2): $G = \{P, H\}$, $\Ganc = \varnothing$.}
\begin{center}
  \groupgraph{grp}{rst}{grp}{rst}{rst}
\end{center}
Strike out the factors with $\Grest$ in the joint distribution:
\[
  p(\vg{P}, \vr{D}, \vr{A}, \vg{H}, \vr{T})
  = p(\vg{P}) \, \strike{p(\vr{D})} \, \strike{p(\vr{A} \mid \vg{P}, \vr{D})} \, p(\vg{H} \mid \vg{P}) \, \strike{p(\vr{T} \mid \vr{A})}.
\]
Keep the other factors:
\[
  p(P, H) = p(P) \, p(H \mid P).
\]

\solstep{Step 3. Find the numerator (Idea, Step~3).}
$\Ganc$ is empty, so there is no sum:
\[
  p(P = 1, H = 1)
  = p(P = 1) \, p(H = 1 \mid P = 1)
  = 0.2 \cdot 0.9
  = 0.18.
\]

\solstep{Step 4. Find $p(G, \Ganc)$ for the denominator (Idea, Step~2): $G = \{H\}$, $\Ganc = \{P\}$.}
\begin{center}
  \groupgraph{anc}{rst}{grp}{rst}{rst}
\end{center}
Strike out the factors with $\Grest$ in the joint distribution:
\[
  p(\va{P}, \vr{D}, \vr{A}, \vg{H}, \vr{T})
  = p(\va{P}) \, \strike{p(\vr{D})} \, \strike{p(\vr{A} \mid \va{P}, \vr{D})} \, p(\vg{H} \mid \va{P}) \, \strike{p(\vr{T} \mid \vr{A})}.
\]
Keep the other factors:
\[
  p(H, P) = p(P) \, p(H \mid P).
\]

\solstep{Step 5. Find the denominator (Idea, Step~3).}
\begin{align*}
  p(H = 1)
  &= \sum_{p} p(P = p) \, p(H = 1 \mid P = p)
    && \text{(Step 4)} \\
  &= 0.2 \cdot 0.9 + (1 - 0.2) \cdot 0.2
   = 0.18 + 0.16
   = 0.34.
\end{align*}

\solstep{Step 6. Divide.}
\[
  p(P = 1 \mid H = 1)
  = \frac{0.18}{0.34}
  = \frac{9}{17}
  \approx 0.529.
\]

\solsubproblem{(b) Find $p(P = 1 \mid T = 1)$.}

\solstep{Step 1. Write the goal as a ratio (Idea, Step~1).}
\[
  p(P = 1 \mid T = 1) = \frac{p(P = 1, T = 1)}{p(T = 1)}.
\]

\solstep{Step 2. Find $p(G, \Ganc)$ for the numerator (Idea, Step~2): $G = \{P, T\}$, $\Ganc = \{A, D\}$.}
\begin{center}
  \groupgraph{grp}{anc}{rst}{anc}{grp}
\end{center}
Strike out the factors with $\Grest$ in the joint distribution:
\[
  p(\vg{P}, \va{D}, \va{A}, \vr{H}, \vg{T})
  = p(\vg{P}) \, p(\va{D}) \, p(\va{A} \mid \vg{P}, \va{D}) \, \strike{p(\vr{H} \mid \vg{P})} \, p(\vg{T} \mid \va{A}).
\]
Keep the other factors:
\[
  p(P, T, A, D) = p(P) \, p(D) \, p(A \mid P, D) \, p(T \mid A).
\]

\solstep{Step 3. Find the numerator (Idea, Step~3).}
\begin{align*}
  &p(P = 1, T = 1) \\
  &= \sum_{a} \sum_{d} p(P = 1) \, p(D = d) \, p(A = a \mid P = 1, D = d) \,
     p(T = 1 \mid A = a)
    && \text{(Step 2)} \\
  &= p(P = 1) \sum_{a} p(T = 1 \mid A = a) \sum_{d}
     p(A = a \mid P = 1, D = d) \, p(D = d)
    && \text{(constants out)} \\
  &= 0.2 \, \bigl[ 0.95 \, \bigl(0.999 \cdot 0.4 + 0.9 \cdot (1 - 0.4)\bigr)
    && \text{($a = 1$, $a = 0$)} \\
  &\qquad + 0.5 \, \bigl((1 - 0.999) \cdot 0.4 + (1 - 0.9) \cdot (1 - 0.4)\bigr)
     \bigr] \\
  &= 0.2 \, [0.95 \cdot 0.9396 + 0.5 \cdot 0.0604] \\
  &= 0.2 \, [0.89262 + 0.0302]
   = 0.2 \cdot 0.92282
   = 0.184564.
\end{align*}

\solstep{Step 4. Find $p(G, \Ganc)$ for the denominator (Idea, Step~2): $G = \{T\}$, $\Ganc = \{P, A, D\}$.}
\begin{center}
  \groupgraph{anc}{anc}{rst}{anc}{grp}
\end{center}
Strike out the factors with $\Grest$ in the joint distribution:
\[
  p(\va{P}, \va{D}, \va{A}, \vr{H}, \vg{T})
  = p(\va{P}) \, p(\va{D}) \, p(\va{A} \mid \va{P}, \va{D}) \, \strike{p(\vr{H} \mid \va{P})} \, p(\vg{T} \mid \va{A}).
\]
Keep the other factors:
\[
  p(T, P, A, D) = p(P) \, p(D) \, p(A \mid P, D) \, p(T \mid A).
\]

\solstep{Step 5. Find the denominator (Idea, Step~3).}
\begin{align*}
  &p(T = 1) \\
  &= \sum_{p} \sum_{a} \sum_{d} p(P = p) \, p(D = d) \,
     p(A = a \mid P = p, D = d) \, p(T = 1 \mid A = a)
    && \text{(Step 4)} \\
  &= \sum_{p} p(P = p) \sum_{a} p(T = 1 \mid A = a) \sum_{d}
     p(A = a \mid P = p, D = d) \, p(D = d)
    && \text{(constants out)} \\
  &= 0.184564 + (1 - 0.2) \, \bigl[ 0.95 \, (0.9 \cdot 0.4 + 0.01 \cdot 0.6)
    && \text{($p = 1$: Step 3)} \\
  &\qquad + 0.5 \, \bigl((1 - 0.9) \cdot 0.4 + (1 - 0.01) \cdot 0.6\bigr) \bigr] \\
  &= 0.184564 + 0.8 \, [0.95 \cdot 0.366 + 0.5 \cdot 0.634] \\
  &= 0.184564 + 0.8 \, [0.3477 + 0.317]
   = 0.184564 + 0.8 \cdot 0.6647 \\
  &= 0.184564 + 0.53176
   = 0.716324.
\end{align*}

\solstep{Step 6. Divide.}
\[
  p(P = 1 \mid T = 1)
  = \frac{0.184564}{0.716324}
  \approx 0.258.
\]

\solsubproblem{(c) Find $p(P = 1 \mid T = 1, H = 1)$.}

\solstep{Step 1. Write the goal as a ratio (Idea, Step~1).}
\[
  p(P = 1 \mid T = 1, H = 1)
  = \frac{p(P = 1, H = 1, T = 1)}{p(H = 1, T = 1)}.
\]

\solstep{Step 2. Find $p(G, \Ganc)$ for the numerator (Idea, Step~2): $G = \{P, H, T\}$, $\Ganc = \{A, D\}$.}
\begin{center}
  \groupgraph{grp}{anc}{grp}{anc}{grp}
\end{center}
Strike out the factors with $\Grest$ in the joint distribution:
\[
  p(\vg{P}, \va{D}, \va{A}, \vg{H}, \vg{T})
  = p(\vg{P}) \, p(\va{D}) \, p(\va{A} \mid \vg{P}, \va{D}) \, p(\vg{H} \mid \vg{P}) \, p(\vg{T} \mid \va{A}).
\]
Keep the other factors:
\[
  p(P, H, T, A, D) = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, p(T \mid A).
\]

\solstep{Step 3. Find the numerator (Idea, Step~3).}
\begin{align*}
  &p(P = 1, H = 1, T = 1) \\
  &= \sum_{a} \sum_{d} p(P = 1) \, p(D = d) \, p(A = a \mid P = 1, D = d)
    && \text{(Step 2)} \\
  &\qquad \cdot p(H = 1 \mid P = 1) \, p(T = 1 \mid A = a) \\
  &= p(H = 1 \mid P = 1)
    && \text{(constants out)} \\
  &\qquad \cdot p(P = 1) \sum_{a} p(T = 1 \mid A = a)
     \sum_{d} p(A = a \mid P = 1, D = d) \, p(D = d) \\
  &= 0.9 \cdot 0.184564
    && \text{((b), Step 3)} \\
  &= 0.1661076.
\end{align*}

\solstep{Step 4. Find $p(G, \Ganc)$ for the denominator (Idea, Step~2): $G = \{H, T\}$, $\Ganc = \{P, A, D\}$.}
\begin{center}
  \groupgraph{anc}{anc}{grp}{anc}{grp}
\end{center}
Strike out the factors with $\Grest$ in the joint distribution:
\[
  p(\va{P}, \va{D}, \va{A}, \vg{H}, \vg{T})
  = p(\va{P}) \, p(\va{D}) \, p(\va{A} \mid \va{P}, \va{D}) \, p(\vg{H} \mid \va{P}) \, p(\vg{T} \mid \va{A}).
\]
Keep the other factors:
\[
  p(H, T, P, A, D) = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, p(T \mid A).
\]

\solstep{Step 5. Find the denominator (Idea, Step~3).}
\begin{align*}
  &p(H = 1, T = 1) \\
  &= \sum_{p} \sum_{a} \sum_{d} p(P = p) \, p(D = d) \,
     p(A = a \mid P = p, D = d)
    && \text{(Step 4)} \\
  &\qquad \cdot p(H = 1 \mid P = p) \, p(T = 1 \mid A = a) \\
  &= \sum_{p} p(H = 1 \mid P = p)
    && \text{(constants out)} \\
  &\qquad \cdot p(P = p) \sum_{a} p(T = 1 \mid A = a)
     \sum_{d} p(A = a \mid P = p, D = d) \, p(D = d) \\
  &= 0.9 \cdot 0.184564 + 0.2 \cdot 0.53176
    && \text{((b), Step 5)} \\
  &= 0.1661076 + 0.106352
   = 0.2724596.
\end{align*}

\solstep{Step 6. Divide.}
\[
  p(P = 1 \mid T = 1, H = 1)
  = \frac{0.1661076}{0.2724596}
  \approx 0.610.
\]

\solpart{Conclusion}
\[
  \underbrace{0.2}_{\text{prior}}
  \;\to\;
  \underbrace{0.529}_{H = 1}, \quad
  \underbrace{0.258}_{T = 1}, \quad
  \underbrace{0.610}_{H = 1,\, T = 1}.
\]
An unhappy teacher is weak evidence: without the party, the teacher is
also often unhappy ($0.6647$ in Step~5 of (b)). Two observations together give the strongest evidence.
\end{solution}
