\label{prob:student-network}
Consider the following probabilistic model. At a seminar, a student answers
badly ($A = 1$) or well ($A = 0$). This depends on whether the student has
depression ($D = 1$ or $D = 0$) and whether the student went to a party the
night before ($P = 1$ or $P = 0$). The party can also give the student a
headache ($H = 1$). A bad answer makes the teacher unhappy ($T = 1$).
Figure~\ref{fig:student-network} shows the causal relations in the model.
The probabilities are:
\begin{center}
  \begin{tabular}[t]{c|cc}
    \toprule
    $p(A = 1 \mid P, D)$ & $P$ & $D$ \\
    \midrule
    0.999 & 1 & 1 \\
    0.9   & 1 & 0 \\
    0.9   & 0 & 1 \\
    0.01  & 0 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(H = 1 \mid P)$ & $P$ \\
    \midrule
    0.9 & 1 \\
    0.2 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(T = 1 \mid A)$ & $A$ \\
    \midrule
    0.95 & 1 \\
    0.5  & 0 \\
    \bottomrule
  \end{tabular}

  \medskip
  $p(P = 1) = 0.2, \qquad p(D = 1) = 0.4.$
\end{center}
Find $p(P = 1 \mid H = 1)$, $p(P = 1 \mid T = 1)$, and
$p(P = 1 \mid T = 1, H = 1)$.

\begin{solution}
\solpart{Idea}
Problem~\ref{prob:medical-test} is a special case of the algorithm below. The
tables give $p(X = 1 \mid \text{parents})$ for each node $X$, and
$p(X = 0 \mid \text{parents}) = 1 - p(X = 1 \mid \text{parents})$.

\emph{Joint distribution.} Apply the chain rule one variable at a time,
from the children to the parents. A missing arrow means no direct
influence: given its parents, a node does not depend on the other nodes in
the condition.
\begin{align*}
  p(P, D, A, H, T)
  &= p(P, D, A, H) \, p(T \mid P, D, A, H)
   && \text{(chain rule)} \\
  &= p(P, D, A, H) \, p(T \mid A)
   && \text{(parent of $T$: $A$)} \\
  p(P, D, A, H)
  &= p(P, D, A) \, p(H \mid P, D, A)
   && \text{(chain rule)} \\
  &= p(P, D, A) \, p(H \mid P)
   && \text{(parent of $H$: $P$)} \\
  p(P, D, A)
  &= p(P, D) \, p(A \mid P, D)
   && \text{(chain rule; parents of $A$: $P$, $D$)} \\
  p(P, D)
  &= p(P) \, p(D \mid P)
   && \text{(chain rule)} \\
  &= p(P) \, p(D)
   && \text{($D$ has no parents)}
\end{align*}
Thus
\[
  p(P, D, A, H, T)
  = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, p(T \mid A).
\]

\emph{Algorithm for $p(V)$}, where $V$ is a set of variables with fixed
values:
\begin{enumerate}[label=\arabic*.]
  \item $\an(V)$: the ancestors of $V$ that are not in $V$. $\Rest$: all
    other variables.
  \item Sum out $\Rest$:
    \begin{align*}
      p(V)
      &= \sum_{\an(V)} \sum_{\Rest} p(V, \an(V), \Rest) \\
      &= \sum_{\an(V)} \sum_{\Rest} p(\Rest \mid V, \an(V)) \,
         p(V, \an(V))
        && \text{(chain rule)} \\
      &= \sum_{\an(V)} p(V, \an(V)) \sum_{\Rest} p(\Rest \mid V, \an(V))
        && \text{(constants out)} \\
      &= \sum_{\an(V)} p(V, \an(V))
        && \textstyle\text{($\sum_{\Rest} p(\Rest \mid V, \an(V)) = 1$)}.
    \end{align*}
  \item $V$ and $\an(V)$ together contain the parents of all their
    variables. Thus the chain rule gives a product of the given tables:
    $p(V, \an(V)) = \prod_{v} p(v \mid \text{parents of } v)$.
  \item Move the factors that do not depend on a summation variable out of
    its sum. Then calculate.
\end{enumerate}
For a query $Q$ and an observation $E$, find $p(Q, E)$ and $p(E)$ with the
algorithm, then divide:
\[
  p(Q \mid E) = \frac{p(Q, E)}{p(E)}.
\]

\solpart{Solution}
\solsubproblem{(a) Find $p(P = 1 \mid H = 1)$.}

\solstep{Step 1. Write the goal as a ratio.}
\[
  p(P = 1 \mid H = 1) = \frac{p(P = 1, H = 1)}{p(H = 1)}.
\]

\solstep{Step 2. Find the numerator: $V = \{P, H\}$, $\an(V) = \varnothing$.}
\[
  p(P = 1, H = 1)
  = p(P = 1) \, p(H = 1 \mid P = 1)
  = 0.2 \cdot 0.9
  = 0.18.
\]

\solstep{Step 3. Find the denominator: $V = \{H\}$, $\an(V) = \{P\}$.}
\begin{align*}
  p(H = 1)
  &= \sum_{p} p(P = p) \, p(H = 1 \mid P = p) \\
  &= 0.2 \cdot 0.9 + (1 - 0.2) \cdot 0.2
   = 0.18 + 0.16
   = 0.34.
\end{align*}

\solstep{Step 4. Divide.}
\[
  p(P = 1 \mid H = 1)
  = \frac{0.18}{0.34}
  = \frac{9}{17}
  \approx 0.529.
\]

\solsubproblem{(b) Find $p(P = 1 \mid T = 1)$.}

\solstep{Step 1. Write the goal as a ratio.}
\[
  p(P = 1 \mid T = 1) = \frac{p(P = 1, T = 1)}{p(T = 1)}.
\]

\solstep{Step 2. Find the numerator: $V = \{P, T\}$, $\an(V) = \{A, D\}$.}
\begin{align*}
  &p(P = 1, T = 1) \\
  &= \sum_{a} \sum_{d} p(P = 1) \, p(D = d) \, p(A = a \mid P = 1, D = d) \,
     p(T = 1 \mid A = a)
    && \text{(algorithm)} \\
  &= p(P = 1) \sum_{a} p(T = 1 \mid A = a) \sum_{d}
     p(A = a \mid P = 1, D = d) \, p(D = d)
    && \text{(constants out)} \\
  &= 0.2 \, \bigl[ 0.95 \, \bigl(0.999 \cdot 0.4 + 0.9 \cdot (1 - 0.4)\bigr)
    && \text{($a = 1$, $a = 0$)} \\
  &\qquad + 0.5 \, \bigl((1 - 0.999) \cdot 0.4 + (1 - 0.9) \cdot (1 - 0.4)\bigr)
     \bigr] \\
  &= 0.2 \, [0.95 \cdot 0.9396 + 0.5 \cdot 0.0604] \\
  &= 0.2 \, [0.89262 + 0.0302]
   = 0.2 \cdot 0.92282
   = 0.184564.
\end{align*}

\solstep{Step 3. Find the denominator: $V = \{T\}$, $\an(V) = \{P, A, D\}$.}
\begin{align*}
  &p(T = 1) \\
  &= \sum_{p} \sum_{a} \sum_{d} p(P = p) \, p(D = d) \,
     p(A = a \mid P = p, D = d) \, p(T = 1 \mid A = a)
    && \text{(algorithm)} \\
  &= \sum_{p} p(P = p) \sum_{a} p(T = 1 \mid A = a) \sum_{d}
     p(A = a \mid P = p, D = d) \, p(D = d)
    && \text{(constants out)} \\
  &= 0.184564 + (1 - 0.2) \, \bigl[ 0.95 \, (0.9 \cdot 0.4 + 0.01 \cdot 0.6)
    && \text{($p = 1$: Step 2)} \\
  &\qquad + 0.5 \, \bigl((1 - 0.9) \cdot 0.4 + (1 - 0.01) \cdot 0.6\bigr) \bigr] \\
  &= 0.184564 + 0.8 \, [0.95 \cdot 0.366 + 0.5 \cdot 0.634] \\
  &= 0.184564 + 0.8 \, [0.3477 + 0.317]
   = 0.184564 + 0.8 \cdot 0.6647 \\
  &= 0.184564 + 0.53176
   = 0.716324.
\end{align*}

\solstep{Step 4. Divide.}
\[
  p(P = 1 \mid T = 1)
  = \frac{0.184564}{0.716324}
  \approx 0.258.
\]

\solsubproblem{(c) Find $p(P = 1 \mid T = 1, H = 1)$.}

\solstep{Step 1. Write the goal as a ratio.}
\[
  p(P = 1 \mid T = 1, H = 1)
  = \frac{p(P = 1, H = 1, T = 1)}{p(H = 1, T = 1)}.
\]

\solstep{Step 2. Find the numerator: $V = \{P, H, T\}$, $\an(V) = \{A, D\}$.}
\begin{align*}
  &p(P = 1, H = 1, T = 1) \\
  &= \sum_{a} \sum_{d} p(P = 1) \, p(D = d) \, p(A = a \mid P = 1, D = d)
    && \text{(algorithm)} \\
  &\qquad \cdot p(H = 1 \mid P = 1) \, p(T = 1 \mid A = a) \\
  &= p(H = 1 \mid P = 1)
    && \text{(constants out)} \\
  &\qquad \cdot p(P = 1) \sum_{a} p(T = 1 \mid A = a)
     \sum_{d} p(A = a \mid P = 1, D = d) \, p(D = d) \\
  &= 0.9 \cdot 0.184564
    && \text{((b), Step 2)} \\
  &= 0.1661076.
\end{align*}

\solstep{Step 3. Find the denominator: $V = \{H, T\}$, $\an(V) = \{P, A, D\}$.}
\begin{align*}
  &p(H = 1, T = 1) \\
  &= \sum_{p} \sum_{a} \sum_{d} p(P = p) \, p(D = d) \,
     p(A = a \mid P = p, D = d)
    && \text{(algorithm)} \\
  &\qquad \cdot p(H = 1 \mid P = p) \, p(T = 1 \mid A = a) \\
  &= \sum_{p} p(H = 1 \mid P = p)
    && \text{(constants out)} \\
  &\qquad \cdot p(P = p) \sum_{a} p(T = 1 \mid A = a)
     \sum_{d} p(A = a \mid P = p, D = d) \, p(D = d) \\
  &= 0.9 \cdot 0.184564 + 0.2 \cdot 0.53176
    && \text{((b), Step 3)} \\
  &= 0.1661076 + 0.106352
   = 0.2724596.
\end{align*}

\solstep{Step 4. Divide.}
\[
  p(P = 1 \mid T = 1, H = 1)
  = \frac{0.1661076}{0.2724596}
  \approx 0.610.
\]

\solpart{Conclusion}
\[
  \underbrace{0.2}_{\text{prior}}
  \;\to\;
  \underbrace{0.529}_{H = 1}, \quad
  \underbrace{0.258}_{T = 1}, \quad
  \underbrace{0.610}_{H = 1,\, T = 1}.
\]
An unhappy teacher is weak evidence: without the party, the teacher is
also often unhappy ($0.6647$ in Step~3 of (b)). Two observations together give the strongest evidence.
\end{solution}
