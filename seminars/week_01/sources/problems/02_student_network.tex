\label{prob:student-network}
Consider the following probabilistic model. At a seminar, a student answers
badly ($A = 1$) or well ($A = 0$). This depends on whether the student has
depression ($D = 1$ or $D = 0$) and whether the student went to a party the
night before ($P = 1$ or $P = 0$). The party can also give the student a
headache ($H = 1$). A bad answer makes the teacher unhappy ($T = 1$).
Figure~\ref{fig:student-network} shows the causal relations in the model.
The probabilities are:
\begin{center}
  \begin{tabular}[t]{c|cc}
    \toprule
    $p(A = 1 \mid P, D)$ & $P$ & $D$ \\
    \midrule
    0.999 & 1 & 1 \\
    0.9   & 1 & 0 \\
    0.9   & 0 & 1 \\
    0.01  & 0 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(H = 1 \mid P)$ & $P$ \\
    \midrule
    0.9 & 1 \\
    0.2 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(T = 1 \mid A)$ & $A$ \\
    \midrule
    0.95 & 1 \\
    0.5  & 0 \\
    \bottomrule
  \end{tabular}

  \medskip
  $p(P = 1) = 0.2, \qquad p(D = 1) = 0.4.$
\end{center}
Find $p(P = 1 \mid H = 1)$, $p(P = 1 \mid T = 1)$, and
$p(P = 1 \mid T = 1, H = 1)$.

  \centering
  \begin{tikzpicture}[
      node distance=0.3cm and 0.5cm,
      var/.style={circle, draw, minimum size=0.75cm},
      >={Stealth},
    ]
    \node[var] (P) {$P$};
    \node[var, right=2cm of P] (D) {$D$};
    \node[var, below left=of P] (H) {$H$};
    \node[var, below right=of P] (A) {$A$};
    \node[var, below=of A] (T) {$T$};
    \draw[->] (P) -- (H);
    \draw[->] (P) -- (A);
    \draw[->] (D) -- (A);
    \draw[->] (A) -- (T);
  \end{tikzpicture}
  \caption{Causal relations for Problem~\ref{prob:student-network}}
  \label{fig:student-network}
\end{figure}

\begin{solution}
\solpart{Idea}
Use the rules of Problem~\ref{prob:medical-test} on a larger graph. The
tables give $p(X = 1 \mid \text{parents})$ for each node $X$, and
$p(X = 0 \mid \text{parents}) = 1 - p(X = 1 \mid \text{parents})$.

\emph{Joint distribution.} Apply the chain rule one variable at a time,
from the children to the parents. A missing arrow means no direct
influence: given its parents, a node does not depend on the other nodes in
the condition.
\begin{align*}
  p(P, D, A, H, T)
  &= p(P, D, A, H) \, p(T \mid P, D, A, H) \\
  &= p(P, D, A, H) \, p(T \mid A)
   && \text{(parent of $T$: $A$)} \\
  p(P, D, A, H)
  &= p(P, D, A) \, p(H \mid P, D, A) \\
  &= p(P, D, A) \, p(H \mid P)
   && \text{(parent of $H$: $P$)} \\
  p(P, D, A)
  &= p(P, D) \, p(A \mid P, D)
   && \text{(parents of $A$: $P$, $D$)} \\
  p(P, D)
  &= p(P) \, p(D \mid P) \\
  &= p(P) \, p(D)
   && \text{($D$ has no parents)}
\end{align*}
Thus
\[
  p(P, D, A, H, T)
  = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, p(T \mid A).
\]

\emph{Rules.} R1 and R2 are the rules of Problem~\ref{prob:medical-test}
for an observation $E$ of a child of $P$. R3--R5 follow from the joint
distribution.
\begin{align*}
  &\text{R1. Against the arrows (Bayes' theorem):}
    && p(P \mid E) = \frac{p(E \mid P) \, p(P)}{p(E)}, \\
  &\text{R2. Marginal probability (chain rule):}
    && p(E) = \sum_{p} p(E \mid P = p) \, p(P = p), \\
  &\text{R3. Unobserved node $Z$ on the path:}
    && p(Y \mid X) = \sum_{z} p(Y \mid Z = z, X) \, p(Z = z \mid X), \\
  &\text{R4. Only the parents matter:}
    && p(T \mid A, P) = p(T \mid A), \quad p(D \mid P) = p(D), \\
  &\text{R5. $H$ and $T$ are independent given $P$:}
    && p(H, T \mid P) = p(H \mid P) \, p(T \mid P).
\end{align*}
R5 follows from the joint distribution:
\begin{align*}
  p(H, T \mid P)
  &= \frac{1}{p(P)} \sum_{a, d}
     p(P) \, p(D = d) \, p(A = a \mid P, D = d) \, p(H \mid P) \,
     p(T \mid A = a) \\
  &= p(H \mid P) \sum_{a} p(T \mid A = a) \sum_{d} p(D = d) \,
     p(A = a \mid P, D = d) \\
  &= p(H \mid P) \sum_{a} p(T \mid A = a) \, p(A = a \mid P)
    && \text{(R3, R4)} \\
  &= p(H \mid P) \, p(T \mid P)
    && \text{(R3, R4)}.
\end{align*}

\solpart{Solution}
In each computation graph, the goal is at the top. Each arrow shows the
rule and the formula for the node above it. Shaded nodes are given.
Dashed nodes are found in an earlier subproblem.

\solsubproblem{(a) Find $p(P = 1 \mid H = 1)$.}
\begin{center}
  \begin{tikzpicture}[cg]
    \node[cg goal]  (G)  at (2, 0)  {$p(P \mid H)$};
    \node[cg node]  (H)  at (2, -2) {$p(H)$};
    \node[cg given] (HP) at (0, -4) {$p(H \mid P)$};
    \node[cg given] (P)  at (7, -4) {$p(P)$};
    \coordinate (g) at (2, -1);
    \draw (H) -- (g);
    \draw (HP) to[out=90, in=180] (g);
    \draw (P) to[out=90, in=0] (g);
    \draw[->] (g) -- node[cg op, right]
      {R1: $p(H \mid P) \, p(P) \,/\, p(H)$} (G);
    \coordinate (h) at (2, -3);
    \draw (HP) -- (h);
    \draw (P) -- (h);
    \draw[->] (h) -- node[cg op, right]
      {R2: $\sum_p p(H \mid P{=}p) \, p(P{=}p)$} (H);
  \end{tikzpicture}
\end{center}

\solstep{Step 1. Find $p(H = 1)$ with R2.}
\begin{align*}
  p(H = 1)
  &= p(H = 1 \mid P = 1) \, p(P = 1) + p(H = 1 \mid P = 0) \, p(P = 0) \\
  &= 0.9 \cdot 0.2 + 0.2 \cdot 0.8
   = 0.18 + 0.16
   = 0.34.
\end{align*}

\solstep{Step 2. Find $p(P = 1 \mid H = 1)$ with R1.}
\[
  p(P = 1 \mid H = 1)
  = \frac{p(H = 1 \mid P = 1) \, p(P = 1)}{p(H = 1)}
  = \frac{0.9 \cdot 0.2}{0.34}
  = \frac{0.18}{0.34}
  = \frac{9}{17}
  \approx 0.529.
\]

\solsubproblem{(b) Find $p(P = 1 \mid T = 1)$.}
\begin{center}
  \begin{tikzpicture}[cg]
    \node[cg goal]  (G)   at (2, 0)    {$p(P \mid T)$};
    \node[cg node]  (T)   at (2, -2)   {$p(T)$};
    \node[cg node]  (TP)  at (0, -4)   {$p(T \mid P)$};
    \node[cg given] (P)   at (7, -4)   {$p(P)$};
    \node[cg given] (TA)  at (-2, -6)  {$p(T \mid A)$};
    \node[cg node]  (AP)  at (1, -6)   {$p(A \mid P)$};
    \node[cg given] (APD) at (0, -8)   {$p(A \mid P, D)$};
    \node[cg given] (D)   at (2.5, -8) {$p(D)$};
    \coordinate (g) at (2, -1);
    \draw (T) -- (g);
    \draw (TP) to[out=90, in=180] (g);
    \draw (P) to[out=90, in=0] (g);
    \draw[->] (g) -- node[cg op, right]
      {R1: $p(T \mid P) \, p(P) \,/\, p(T)$} (G);
    \coordinate (t) at (2, -3);
    \draw (TP) -- (t);
    \draw (P) -- (t);
    \draw[->] (t) -- node[cg op, right]
      {R2: $\sum_p p(T \mid P{=}p) \, p(P{=}p)$} (T);
    \coordinate (tp) at (0, -5);
    \draw (TA) -- (tp);
    \draw (AP) -- (tp);
    \draw[->] (tp) -- node[cg op, left]
      {R3, R4: $\sum_a p(T \mid A{=}a) \, p(A{=}a \mid P)$} (TP);
    \coordinate (ap) at (1, -7);
    \draw (APD) -- (ap);
    \draw (D) -- (ap);
    \draw[->] (ap) -- node[cg op, right]
      {R3, R4: $\sum_d p(A \mid P, D{=}d) \, p(D{=}d)$} (AP);
  \end{tikzpicture}
\end{center}

\solstep{Step 1. Find $p(A = 1 \mid P)$ with R3 and R4.}
\begin{align*}
  p(A = 1 \mid P = 1)
  &= p(A = 1 \mid P = 1, D = 1) \, p(D = 1)
   + p(A = 1 \mid P = 1, D = 0) \, p(D = 0) \\
  &= 0.999 \cdot 0.4 + 0.9 \cdot 0.6
   = 0.3996 + 0.54
   = 0.9396, \\
  p(A = 1 \mid P = 0)
  &= p(A = 1 \mid P = 0, D = 1) \, p(D = 1)
   + p(A = 1 \mid P = 0, D = 0) \, p(D = 0) \\
  &= 0.9 \cdot 0.4 + 0.01 \cdot 0.6
   = 0.36 + 0.006
   = 0.366.
\end{align*}
Thus $p(A = 0 \mid P = 1) = 1 - 0.9396 = 0.0604$ and
$p(A = 0 \mid P = 0) = 1 - 0.366 = 0.634$.

\solstep{Step 2. Find $p(T = 1 \mid P)$ with R3 and R4.}
\begin{align*}
  p(T = 1 \mid P = 1)
  &= p(T = 1 \mid A = 1) \, p(A = 1 \mid P = 1)
   + p(T = 1 \mid A = 0) \, p(A = 0 \mid P = 1) \\
  &= 0.95 \cdot 0.9396 + 0.5 \cdot 0.0604
   = 0.89262 + 0.0302
   = 0.92282, \\
  p(T = 1 \mid P = 0)
  &= p(T = 1 \mid A = 1) \, p(A = 1 \mid P = 0)
   + p(T = 1 \mid A = 0) \, p(A = 0 \mid P = 0) \\
  &= 0.95 \cdot 0.366 + 0.5 \cdot 0.634
   = 0.3477 + 0.317
   = 0.6647.
\end{align*}

\solstep{Step 3. Find $p(T = 1)$ with R2.}
\begin{align*}
  p(T = 1)
  &= p(T = 1 \mid P = 1) \, p(P = 1) + p(T = 1 \mid P = 0) \, p(P = 0) \\
  &= 0.92282 \cdot 0.2 + 0.6647 \cdot 0.8
   = 0.184564 + 0.53176
   = 0.716324.
\end{align*}

\solstep{Step 4. Find $p(P = 1 \mid T = 1)$ with R1.}
\[
  p(P = 1 \mid T = 1)
  = \frac{p(T = 1 \mid P = 1) \, p(P = 1)}{p(T = 1)}
  = \frac{0.92282 \cdot 0.2}{0.716324}
  = \frac{0.184564}{0.716324}
  \approx 0.258.
\]

\solsubproblem{(c) Find $p(P = 1 \mid T = 1, H = 1)$.}
\begin{center}
  \begin{tikzpicture}[cg]
    \node[cg goal]   (G)   at (2, 0)    {$p(P \mid H, T)$};
    \node[cg node]   (HT)  at (2, -2)   {$p(H, T)$};
    \node[cg node]   (HTP) at (0, -4)   {$p(H, T \mid P)$};
    \node[cg given]  (P)   at (7.5, -4) {$p(P)$};
    \node[cg given]  (HP)  at (-1, -6)  {$p(H \mid P)$};
    \node[cg reused] (TP)  at (1, -6)   {$p(T \mid P)$};
    \coordinate (g) at (2, -1);
    \draw (HT) -- (g);
    \draw (HTP) to[out=90, in=180] (g);
    \draw (P) to[out=90, in=0] (g);
    \draw[->] (g) -- node[cg op, right]
      {R1: $p(H, T \mid P) \, p(P) \,/\, p(H, T)$} (G);
    \coordinate (ht) at (2, -3);
    \draw (HTP) -- (ht);
    \draw (P) -- (ht);
    \draw[->] (ht) -- node[cg op, right]
      {R2: $\sum_p p(H, T \mid P{=}p) \, p(P{=}p)$} (HT);
    \coordinate (htp) at (0, -5);
    \draw (HP) -- (htp);
    \draw (TP) -- (htp);
    \draw[->] (htp) -- node[cg op, left]
      {R5: $p(H \mid P) \, p(T \mid P)$} (HTP);
  \end{tikzpicture}
\end{center}
The node $p(T \mid P)$ is from Step~2 of (b).

\solstep{Step 1. Find $p(H = 1, T = 1 \mid P)$ with R5.}
\begin{align*}
  p(H = 1, T = 1 \mid P = 1)
  &= p(H = 1 \mid P = 1) \, p(T = 1 \mid P = 1)
   = 0.9 \cdot 0.92282
   = 0.830538, \\
  p(H = 1, T = 1 \mid P = 0)
  &= p(H = 1 \mid P = 0) \, p(T = 1 \mid P = 0)
   = 0.2 \cdot 0.6647
   = 0.13294.
\end{align*}

\solstep{Step 2. Find $p(H = 1, T = 1)$ with R2.}
\begin{align*}
  p(H = 1, T = 1)
  &= p(H = 1, T = 1 \mid P = 1) \, p(P = 1)
   + p(H = 1, T = 1 \mid P = 0) \, p(P = 0) \\
  &= 0.830538 \cdot 0.2 + 0.13294 \cdot 0.8
   = 0.1661076 + 0.106352
   = 0.2724596.
\end{align*}

\solstep{Step 3. Find $p(P = 1 \mid T = 1, H = 1)$ with R1.}
\[
  p(P = 1 \mid T = 1, H = 1)
  = \frac{p(H = 1, T = 1 \mid P = 1) \, p(P = 1)}{p(H = 1, T = 1)}
  = \frac{0.1661076}{0.2724596}
  \approx 0.610.
\]

\solpart{Conclusion}
\[
  \underbrace{0.2}_{\text{prior}}
  \;\to\;
  \underbrace{0.529}_{H = 1}, \quad
  \underbrace{0.258}_{T = 1}, \quad
  \underbrace{0.610}_{H = 1,\, T = 1}.
\]
An unhappy teacher is weak evidence: $p(T = 1 \mid P = 0) = 0.6647$ is
also high. Two observations together give the strongest evidence.
\end{solution}
