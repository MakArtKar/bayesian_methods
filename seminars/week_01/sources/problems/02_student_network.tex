\label{prob:student-network}
Consider the following probabilistic model. At a seminar, a student answers
badly ($A = 1$) or well ($A = 0$). This depends on whether the student has
depression ($D = 1$ or $D = 0$) and whether the student went to a party the
night before ($P = 1$ or $P = 0$). The party can also give the student a
headache ($H = 1$). A bad answer makes the teacher unhappy ($T = 1$).
Figure~\ref{fig:student-network} shows the causal relations in the model.
The probabilities are:
\begin{center}
  \begin{tabular}[t]{c|cc}
    \toprule
    $p(A = 1 \mid P, D)$ & $P$ & $D$ \\
    \midrule
    0.999 & 1 & 1 \\
    0.9   & 1 & 0 \\
    0.9   & 0 & 1 \\
    0.01  & 0 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(H = 1 \mid P)$ & $P$ \\
    \midrule
    0.9 & 1 \\
    0.2 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(T = 1 \mid A)$ & $A$ \\
    \midrule
    0.95 & 1 \\
    0.5  & 0 \\
    \bottomrule
  \end{tabular}

  \medskip
  $p(P = 1) = 0.2, \qquad p(D = 1) = 0.4.$
\end{center}
Find $p(P = 1 \mid H = 1)$, $p(P = 1 \mid T = 1)$, and
$p(P = 1 \mid T = 1, H = 1)$.

\begin{figure}[ht]
  \centering
  \begin{tikzpicture}[
      node distance=0.3cm and 0.5cm,
      var/.style={circle, draw, minimum size=0.75cm},
      >={Stealth},
    ]
    \node[var] (P) {$P$};
    \node[var, right=2cm of P] (D) {$D$};
    \node[var, below left=of P] (H) {$H$};
    \node[var, below right=of P] (A) {$A$};
    \node[var, below=of A] (T) {$T$};
    \draw[->] (P) -- (H);
    \draw[->] (P) -- (A);
    \draw[->] (D) -- (A);
    \draw[->] (A) -- (T);
  \end{tikzpicture}
  \caption{Causal relations for Problem~\ref{prob:student-network}}
  \label{fig:student-network}
\end{figure}

\begin{solution}
\solpart{Idea}
Use the method of Problem~\ref{prob:medical-test} on a larger graph.

\emph{Joint distribution.} Apply the chain rule one variable at a time,
from the children to the parents. A missing arrow means no direct
influence: given its parents, a node does not depend on the other nodes in
the condition.
\begin{align*}
  p(P, D, A, H, T)
  &= p(P, D, A, H) \, p(T \mid P, D, A, H) \\
  &= p(P, D, A, H) \, p(T \mid A)
   && \text{(parent of $T$: $A$)} \\
  p(P, D, A, H)
  &= p(P, D, A) \, p(H \mid P, D, A) \\
  &= p(P, D, A) \, p(H \mid P)
   && \text{(parent of $H$: $P$)} \\
  p(P, D, A)
  &= p(P, D) \, p(A \mid P, D)
   && \text{(parents of $A$: $P$, $D$)} \\
  p(P, D)
  &= p(P) \, p(D \mid P) \\
  &= p(P) \, p(D)
   && \text{($D$ has no parents)}
\end{align*}
Thus
\[
  p(P, D, A, H, T)
  = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, p(T \mid A).
\]
Each factor is a given table.

\emph{Queries.} Move along the edges with sums, and against the edges with
Bayes' theorem:
\[
  p(T \mid P) = \sum_{A} p(T \mid A) \, p(A \mid P),
  \qquad
  p(P \mid T) = \frac{p(P) \, p(T \mid P)}{\sum_{P'} p(P') \, p(T \mid P')}.
\]
A sum over a node without children removes its factor, because
$\sum_{x} p(x \mid \text{parents}) = 1$.

\solpart{Solution}
\solstep{Step 1. Find $p(P = 1 \mid H = 1)$.}
\[
  p(P = 1 \mid H = 1)
  = \frac{p(P = 1, H = 1)}{p(H = 1)}
  = \frac{p(P = 1, H = 1)}{p(P = 1, H = 1) + p(P = 0, H = 1)}.
\]

\solstep{Step 1a. Find $p(P, H = 1)$ for Step 1.}
Sum out $T$, $A$, $D$ (each factor sums to 1):
\begin{align*}
  p(P, H) &= p(P) \, p(H \mid P), \\
  p(P = 1, H = 1) &= 0.2 \cdot 0.9 = 0.18, \\
  p(P = 0, H = 1) &= (1 - 0.2) \cdot 0.2 = 0.8 \cdot 0.2 = 0.16.
\end{align*}

\solstep{Step 1b. Substitute Step 1a into Step 1.}
\[
  p(P = 1 \mid H = 1)
  = \frac{0.18}{0.18 + 0.16} = \frac{0.18}{0.34} = \frac{9}{17}
  \approx 0.529.
\]

\solstep{Step 2. Find $p(P = 1 \mid T = 1)$.}
\[
  p(P = 1 \mid T = 1)
  = \frac{p(P = 1, T = 1)}{p(P = 1, T = 1) + p(P = 0, T = 1)},
  \qquad
  p(P, T) = p(P) \, p(T \mid P).
\]

\solstep{Step 2a. Find $p(T = 1 \mid P)$ for Step 2.}
\begin{align*}
  p(T \mid P)
  &= \sum_{A} p(T, A \mid P)
   = \sum_{A} p(T \mid A, P) \, p(A \mid P) \\
  &= \sum_{A} p(T \mid A) \, p(A \mid P)
   && \text{(parent of $T$: $A$)}.
\end{align*}

\solstep{Step 2b. Find $p(A = 1 \mid P)$ for Step 2a.}
\begin{align*}
  p(A \mid P)
  &= \sum_{D} p(A, D \mid P)
   = \sum_{D} p(A \mid P, D) \, p(D \mid P) \\
  &= \sum_{D} p(A \mid P, D) \, p(D)
   && \text{($D$ has no parents)}.
\end{align*}
\begin{align*}
  p(A = 1 \mid P = 1) &= 0.999 \cdot 0.4 + 0.9 \cdot (1 - 0.4)
    = 0.3996 + 0.54 = 0.9396, \\
  p(A = 1 \mid P = 0) &= 0.9 \cdot 0.4 + 0.01 \cdot (1 - 0.4)
    = 0.36 + 0.006 = 0.366.
\end{align*}

\solstep{Step 2c. Substitute Step 2b into Step 2a.}
\begin{align*}
  p(T = 1 \mid P = 1) &= 0.95 \cdot 0.9396 + 0.5 \cdot (1 - 0.9396)
    = 0.89262 + 0.0302 = 0.92282, \\
  p(T = 1 \mid P = 0) &= 0.95 \cdot 0.366 + 0.5 \cdot (1 - 0.366)
    = 0.3477 + 0.317 = 0.6647.
\end{align*}

\solstep{Step 2d. Substitute Step 2c into Step 2.}
\begin{align*}
  p(P = 1, T = 1) &= 0.2 \cdot 0.92282 = 0.184564, \\
  p(P = 0, T = 1) &= 0.8 \cdot 0.6647 = 0.53176, \\
  p(P = 1 \mid T = 1)
    &= \frac{0.184564}{0.184564 + 0.53176}
     = \frac{0.184564}{0.716324}
     \approx 0.258.
\end{align*}

\solstep{Step 3. Find $p(P = 1 \mid T = 1, H = 1)$.}
\[
  p(P = 1 \mid T = 1, H = 1)
  = \frac{p(P = 1, H = 1, T = 1)}
         {p(P = 1, H = 1, T = 1) + p(P = 0, H = 1, T = 1)}.
\]

\solstep{Step 3a. Find $p(P, H, T)$ for Step 3.}
\begin{align*}
  p(P, H, T)
  &= \sum_{A} \sum_{D} p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \,
     p(T \mid A) \\
  &= p(P) \, p(H \mid P) \sum_{A} p(T \mid A) \sum_{D} p(D) \,
     p(A \mid P, D)
     && \text{(move factors out of the sums)} \\
  &= p(P) \, p(H \mid P) \sum_{A} p(T \mid A) \, p(A \mid P)
     && \text{(Step 2b)} \\
  &= p(P) \, p(H \mid P) \, p(T \mid P)
     && \text{(Step 2a)}.
\end{align*}
With the values from Step 2c:
\begin{align*}
  p(P = 1, H = 1, T = 1) &= 0.2 \cdot 0.9 \cdot 0.92282 = 0.166108, \\
  p(P = 0, H = 1, T = 1) &= 0.8 \cdot 0.2 \cdot 0.6647 = 0.106352.
\end{align*}

\solstep{Step 3b. Substitute Step 3a into Step 3.}
\[
  p(P = 1 \mid T = 1, H = 1)
  = \frac{0.166108}{0.166108 + 0.106352}
  = \frac{0.166108}{0.27246}
  \approx 0.610.
\]

\solpart{Conclusion}
\[
  \underbrace{0.2}_{\text{prior}}
  \;\to\;
  \underbrace{0.529}_{H = 1}, \quad
  \underbrace{0.258}_{T = 1}, \quad
  \underbrace{0.610}_{H = 1,\, T = 1}.
\]
An unhappy teacher is weak evidence: $p(T = 1 \mid P = 0) = 0.6647$ is
also high. Two observations together give the strongest evidence.
\end{solution}
