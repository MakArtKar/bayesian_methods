\label{prob:student-network}
Consider the following probabilistic model. At a seminar, a student answers
badly ($A = 1$) or well ($A = 0$). This depends on whether the student has
depression ($D = 1$ or $D = 0$) and whether the student went to a party the
night before ($P = 1$ or $P = 0$). The party can also give the student a
headache ($H = 1$). A bad answer makes the teacher unhappy ($T = 1$).
Figure~\ref{fig:student-network} shows the causal relations in the model.
The probabilities are:
\begin{center}
  \begin{tabular}[t]{c|cc}
    \toprule
    $p(A = 1 \mid P, D)$ & $P$ & $D$ \\
    \midrule
    0.999 & 1 & 1 \\
    0.9   & 1 & 0 \\
    0.9   & 0 & 1 \\
    0.01  & 0 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(H = 1 \mid P)$ & $P$ \\
    \midrule
    0.9 & 1 \\
    0.2 & 0 \\
    \bottomrule
  \end{tabular}
  \quad
  \begin{tabular}[t]{c|c}
    \toprule
    $p(T = 1 \mid A)$ & $A$ \\
    \midrule
    0.95 & 1 \\
    0.5  & 0 \\
    \bottomrule
  \end{tabular}

  \medskip
  $p(P = 1) = 0.2, \qquad p(D = 1) = 0.4.$
\end{center}
Find $p(P = 1 \mid H = 1)$, $p(P = 1 \mid T = 1)$, and
$p(P = 1 \mid T = 1, H = 1)$.

  \centering

  \label{fig:student-network}
\end{figure}

\begin{solution}
\solpart{Idea}
Use the plan of Problem~\ref{prob:medical-test} on a larger graph. The
tables give $p(X = 1 \mid \text{parents})$ for each node $X$, and
$p(X = 0 \mid \text{parents}) = 1 - p(X = 1 \mid \text{parents})$.

\emph{Joint distribution.} Apply the chain rule one variable at a time,
from the children to the parents. A missing arrow means no direct
influence: given its parents, a node does not depend on the other nodes in
the condition.
\begin{align*}
  p(P, D, A, H, T)
  &= p(P, D, A, H) \, p(T \mid P, D, A, H) \\
  &= p(P, D, A, H) \, p(T \mid A)
   && \text{(parent of $T$: $A$)} \\
  p(P, D, A, H)
  &= p(P, D, A) \, p(H \mid P, D, A) \\
  &= p(P, D, A) \, p(H \mid P)
   && \text{(parent of $H$: $P$)} \\
  p(P, D, A)
  &= p(P, D) \, p(A \mid P, D)
   && \text{(parents of $A$: $P$, $D$)} \\
  p(P, D)
  &= p(P) \, p(D \mid P) \\
  &= p(P) \, p(D)
   && \text{($D$ has no parents)}
\end{align*}
Thus
\[
  p(P, D, A, H, T)
  = p(P) \, p(D) \, p(A \mid P, D) \, p(H \mid P) \, p(T \mid A).
\]

\emph{Sum out variables without children in the result.} Sum the joint
distribution over a set of variables $Y$, and keep the variables $X$. If no
variable in $X$ has a parent in $Y$, the factors of $Y$ sum to 1:
\[
  p(X) = \prod_{i} p(X_i \mid \text{parents of } X_i).
\]
The reason: sum first over the last variable of $Y$ in the order of the
chain rule. No other factor contains it, and its factor sums to 1. Then
repeat for the other variables of $Y$.

\solpart{Solution}
\solsubproblem{(a) Find $p(P = 1 \mid H = 1)$.}

\solstep{Step 1. Write the goal as a ratio.}
\[
  p(P = 1 \mid H = 1) = \frac{p(P = 1, H = 1)}{p(H = 1)}.
\]

\solstep{Step 2. Find $p(P = p, H = 1)$.}
Sum out $D$, $A$, and $T$. The parents of $P$ and $H$ are not in
$\{D, A, T\}$, so their factors sum to 1:
\[
  p(P = p, H = 1) = p(P = p) \, p(H = 1 \mid P = p).
\]
\begin{align*}
  p(P = 1, H = 1) &= 0.2 \cdot 0.9 = 0.18, \\
  p(P = 0, H = 1) &= (1 - 0.2) \cdot 0.2 = 0.8 \cdot 0.2 = 0.16.
\end{align*}

\solstep{Step 3. Find the denominator.}
\[
  p(H = 1)
  = \sum_{p} p(P = p, H = 1)
  = 0.18 + 0.16
  = 0.34.
\]

\solstep{Step 4. Divide.}
\[
  p(P = 1 \mid H = 1)
  = \frac{0.18}{0.34}
  = \frac{9}{17}
  \approx 0.529.
\]

\solsubproblem{(b) Find $p(P = 1 \mid T = 1)$.}

\solstep{Step 1. Write the goal as a ratio.}
\[
  p(P = 1 \mid T = 1) = \frac{p(P = 1, T = 1)}{p(T = 1)}.
\]

\solstep{Step 2. Find $p(P = p, T = 1)$.}
Sum out $H$: no other variable has the parent $H$. Then sum over $D$ inside
the sum over $A$:
\begin{align*}
  &p(P = p, T = 1) \\
  &= \sum_{a} \sum_{d} p(P = p) \, p(D = d) \,
     p(A = a \mid P = p, D = d) \, p(T = 1 \mid A = a)
    && \text{(sum out $H$)} \\
  &= p(P = p) \sum_{a} p(T = 1 \mid A = a) \sum_{d}
     p(A = a \mid P = p, D = d) \, p(D = d)
    && \text{(constants out)} \\
  &= p(P = p) \sum_{a} p(T = 1 \mid A = a) \, p(A = a \mid P = p)
    && \text{(sum over $D$)} \\
  &= p(P = p) \, p(T = 1 \mid P = p)
    && \text{(sum over $A$)}.
\end{align*}
The sum over $D$ uses $p(D = d) = p(D = d \mid P = p)$: the sum-out rule
gives $p(P, D) = p(P) \, p(D)$. Thus
$\sum_{d} p(A = a \mid P = p, D = d) \, p(D = d \mid P = p)
= \sum_{d} p(A = a, D = d \mid P = p) = p(A = a \mid P = p)$.
The sum over $A$ is the same step with $T$ and $A$.

\solstep{Step 3. Find $p(A = 1 \mid P = p)$ for Step 2.}
\begin{align*}
  p(A = 1 \mid P = 1)
  &= \sum_{d} p(A = 1 \mid P = 1, D = d) \, p(D = d)
   = 0.999 \cdot 0.4 + 0.9 \cdot (1 - 0.4) \\
  &= 0.3996 + 0.54
   = 0.9396, \\
  p(A = 1 \mid P = 0)
  &= \sum_{d} p(A = 1 \mid P = 0, D = d) \, p(D = d)
   = 0.9 \cdot 0.4 + 0.01 \cdot (1 - 0.4) \\
  &= 0.36 + 0.006
   = 0.366.
\end{align*}

\solstep{Step 4. Find $p(T = 1 \mid P = p)$ for Step 2.}
\begin{align*}
  p(T = 1 \mid P = 1)
  &= \sum_{a} p(T = 1 \mid A = a) \, p(A = a \mid P = 1)
   = 0.95 \cdot 0.9396 + 0.5 \cdot (1 - 0.9396) \\
  &= 0.89262 + 0.0302
   = 0.92282, \\
  p(T = 1 \mid P = 0)
  &= \sum_{a} p(T = 1 \mid A = a) \, p(A = a \mid P = 0)
   = 0.95 \cdot 0.366 + 0.5 \cdot (1 - 0.366) \\
  &= 0.3477 + 0.317
   = 0.6647.
\end{align*}

\solstep{Step 5. Find the numerator and the denominator.}
\begin{align*}
  p(P = 1, T = 1) &= 0.2 \cdot 0.92282 = 0.184564, \\
  p(P = 0, T = 1) &= 0.8 \cdot 0.6647 = 0.53176, \\
  p(T = 1) &= \sum_{p} p(P = p, T = 1) = 0.184564 + 0.53176 = 0.716324.
\end{align*}

\solstep{Step 6. Divide.}
\[
  p(P = 1 \mid T = 1)
  = \frac{0.184564}{0.716324}
  \approx 0.258.
\]

\solsubproblem{(c) Find $p(P = 1 \mid T = 1, H = 1)$.}

\solstep{Step 1. Write the goal as a ratio.}
\[
  p(P = 1 \mid T = 1, H = 1)
  = \frac{p(P = 1, H = 1, T = 1)}{p(H = 1, T = 1)}.
\]

\solstep{Step 2. Find $p(P = p, H = 1, T = 1)$.}
Sum over $D$ and $A$ as in Step~2 of (b). The factor $p(H = 1 \mid P = p)$
is a constant:
\begin{align*}
  &p(P = p, H = 1, T = 1) \\
  &= \sum_{a} \sum_{d} p(P = p) \, p(D = d) \,
     p(A = a \mid P = p, D = d) \\
  &\qquad \cdot p(H = 1 \mid P = p) \, p(T = 1 \mid A = a) \\
  &= p(P = p) \, p(H = 1 \mid P = p) \\
  &\qquad \cdot \sum_{a} p(T = 1 \mid A = a)
     \sum_{d} p(A = a \mid P = p, D = d) \, p(D = d)
    && \text{(constants out)} \\
  &= p(P = p) \, p(H = 1 \mid P = p) \, p(T = 1 \mid P = p)
    && \text{(Step 2 of (b))}.
\end{align*}
With Step~4 of (b):
\begin{align*}
  p(P = 1, H = 1, T = 1) &= 0.2 \cdot 0.9 \cdot 0.92282
    = 0.18 \cdot 0.92282 = 0.1661076, \\
  p(P = 0, H = 1, T = 1) &= 0.8 \cdot 0.2 \cdot 0.6647
    = 0.16 \cdot 0.6647 = 0.106352.
\end{align*}

\solstep{Step 3. Find the denominator.}
\[
  p(H = 1, T = 1)
  = \sum_{p} p(P = p, H = 1, T = 1)
  = 0.1661076 + 0.106352
  = 0.2724596.
\]

\solstep{Step 4. Divide.}
\[
  p(P = 1 \mid T = 1, H = 1)
  = \frac{0.1661076}{0.2724596}
  \approx 0.610.
\]

\solpart{Conclusion}
\[
  \underbrace{0.2}_{\text{prior}}
  \;\to\;
  \underbrace{0.529}_{H = 1}, \quad
  \underbrace{0.258}_{T = 1}, \quad
  \underbrace{0.610}_{H = 1,\, T = 1}.
\]
An unhappy teacher is weak evidence: $p(T = 1 \mid P = 0) = 0.6647$ is
also high. Two observations together give the strongest evidence.
\end{solution}
